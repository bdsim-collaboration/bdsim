89G4VParticleChange* BDSLaserCumulativeCompton::PostStepDoIt(const G4Track& track,
90 const G4Step& step)
91{
92 aParticleChange.Initialize(track);
93 G4LogicalVolume* lv = track.GetVolume()->GetLogicalVolume();
94 if (!lv->IsExtended()) // not extended so can't be a laser logical volume
95 {return pParticleChange;}
96 BDSLogicalVolumeLaser* lvv = dynamic_cast<BDSLogicalVolumeLaser*>(lv);
97 if (!lvv)// it's an extended volume but not ours (could be a crystal)
98 {return pParticleChange;}
99 // else proceed
100
102 BDSUserTrackInformation* trackInfo = dynamic_cast<BDSUserTrackInformation*>(track.GetUserInformation());
103 if (!trackInfo)
104 {return pParticleChange;}
105 else if (trackInfo->GetComptonScattered())
106 {return pParticleChange;}
107 else
108 {
109 // ############ get particle info
110 const G4DynamicParticle* particle = track.GetDynamicParticle();
111 G4double particleEnergy = particle->GetTotalEnergy();
112 G4ThreeVector particleMomentum = particle->GetMomentum();
113 G4ThreeVector particlePolarization = trackInfo->GetPolarizationState()->GetPolarization();
114
115 G4ThreeVector particleBeta = particleMomentum/particleEnergy;
116 G4double particleGamma = particleEnergy/particle->GetMass();
117 G4double particleVelocity = particleBeta.mag()*CLHEP::c_light;
118 G4LorentzVector particle4VectorMomentum = particle->Get4Momentum();
119 G4int partID = particle->GetParticleDefinition()->GetPDGEncoding();
120
121 G4double particleTimePostStepGlobal = track.GetGlobalTime();
122 G4double particleTimePreStepGlobal = step.GetPreStepPoint()->GetGlobalTime();
123
124 //######### Get particle position and momentum direction ###############################
125 G4ThreeVector particlePositionPostStepGlobal = track.GetPosition();
126 G4ThreeVector particlePositionPreStepGlobal = track.GetStep()->GetPreStepPoint()->GetPosition();
127 G4ThreeVector particleMomentumDirectionGlobal = track.GetStep()->GetPreStepPoint()->GetMomentumDirection();
128 G4ThreeVector stepVector = particlePositionPostStepGlobal-particlePositionPreStepGlobal;
129 G4double stepMagnitude = stepVector.mag();
130
131 //#################### Get rotation and translation matrices for global to laser coordinates ############################
132
133 const G4RotationMatrix* rot = track.GetTouchable()->GetRotation();
134 const G4AffineTransform transform = track.GetTouchable()->GetHistory()->GetTopTransform();
135
136 // ############### Need local pre step points to step through laser ###############
137 G4ThreeVector particlePositionLocal = transform.TransformPoint(particlePositionPreStepGlobal);
138 G4ThreeVector particleMomentumDirectionLocal = transform.TransformPoint(particleMomentumDirectionGlobal).unit();
139
140 //########################## Get Laser #####################################
141 const BDSLaser* laser = lvv->Laser();
142 //######################## Get/Create photon information ##############################
143 G4ThreeVector photonPolarization = laser->Polarization();
144 G4ThreeVector photonUnit(0,0,1);
145
146 // ################ Rotate the photon information to the correct orientation ###########
147 photonPolarization.transform(*rot);
148 particlePolarization.transform(*rot);
149 photonUnit.transform(*rot);
150 G4double photonEnergy = (CLHEP::h_Planck*CLHEP::c_light)/laser->Wavelength();
151
152 // ################ Create a Lorentz Vector for the photon for easy boosting #################
153 G4ThreeVector photonVector = photonUnit*photonEnergy;
154 G4LorentzVector photonLorentz = G4LorentzVector(photonVector,photonEnergy);
155
156 // ################ boost to rest frame #################
157 photonLorentz.boost(-1.0*particleBeta);
158 particle4VectorMomentum.boost(-particleBeta);
159 G4double photonEnergyLorentz = photonLorentz.e();
160
161 //#################### create array and value for the total laser flux
162 G4double photonFluxSum = 0;
163 std::vector<G4double> fluxArray;
164
165 std::vector<G4LorentzVector> trajectoryPositions;
166
167 for(G4int i = 0;i<=99;i++)
168 {
169 G4ThreeVector stepPositionGlobal = particlePositionPreStepGlobal+float(i)*(stepMagnitude/100.)*particleMomentumDirectionGlobal;
170 G4ThreeVector stepPositionLocal = transform.TransformPoint(stepPositionGlobal);
171 G4double particleStepGlobalTime = particleTimePreStepGlobal+((float(i)*(stepMagnitude/100.))/particleVelocity);
172 G4double stepIntensity = ((laser->Intensity(stepPositionLocal)/photonEnergyLorentz)
173 * laser->TemporalProfileGaussian(particleStepGlobalTime,stepPositionLocal.z()));
174 photonFluxSum = photonFluxSum + stepIntensity;
175 fluxArray.push_back(stepIntensity);
176 }
177
178 G4double crossSection = comptonEngine->CrossSection(photonEnergyLorentz,partID);
179
180 G4double stepTime = ((particleTimePostStepGlobal-particleTimePreStepGlobal)*particleGamma)/100.;//divide by 100 for the 100 steps taken
181 G4double cumulativeProbability = 1.0 - std::exp(-1.0*crossSection*photonFluxSum*stepTime);
182 G4double secondaryStepPosition;
183
184 if(photonFluxSum == 0)
185 {secondaryStepPosition = G4UniformRand();}
186 else
187 {
188 G4RandGeneral trajectoryPDFRandom = CLHEP::RandGeneral(*CLHEP::HepRandom::getTheEngine(),
189 fluxArray.data(),100,0);
190 secondaryStepPosition = trajectoryPDFRandom.shoot();
191 }
192
193 G4ThreeVector proposedPositionGlobal = particlePositionPreStepGlobal + (stepMagnitude*secondaryStepPosition*particleMomentumDirectionGlobal);
194 G4double proposedTime = particleTimePreStepGlobal + (stepMagnitude*secondaryStepPosition/particleVelocity);
195
196 aParticleChange.ProposePosition(proposedPositionGlobal);
197 G4double initialWeight=aParticleChange.GetParentWeight();
198 aParticleChange.ProposeParentWeight(initialWeight*cumulativeProbability);
199 aParticleChange.SetNumberOfSecondaries(1);
200 comptonEngine->setIncomingElectron(particle4VectorMomentum);
201 comptonEngine->setIncomingGamma(photonLorentz);
202 comptonEngine->SetIncomingGammaPolarization(G4StokesVector(photonPolarization));
203 comptonEngine->SetIncomingElectronPolarization(G4StokesVector(particlePolarization));
204 comptonEngine->PerformCompton(particleBeta,partID);
205 G4LorentzVector scatteredGamma = comptonEngine->GetScatteredGamma();
206 G4DynamicParticle* gamma = new G4DynamicParticle(G4Gamma::Gamma(),
207 scatteredGamma.vect().unit(),// direction
208 scatteredGamma.e());
209
210 G4LorentzVector scatteredParticle = comptonEngine->GetScatteredElectron();
211 G4LorentzVector particleLorentz = G4LorentzVector(scatteredParticle.vect().unit(),scatteredParticle.e());
212 aParticleChange.AddSecondary(gamma,proposedTime);
213 aParticleChange.ProposeEnergy(particleLorentz.e());
214 aParticleChange.ProposeMomentumDirection(particleLorentz.getX(),particleLorentz.getY(),particleLorentz.getZ());
215 aParticleChange.ProposePosition(particlePositionPreStepGlobal + particleLorentz.vect()*stepVector.mag());
216 trackInfo->setComptonScatteredTrue();
217 return G4VDiscreteProcess::PostStepDoIt(track, step);
218 }
219}
