82BDSBeamPipeFactoryBase* BDSBeamPipeFactory::GetAppropriateFactory(BDSBeamPipeType type)
83{
84 switch(type.underlying())
85 {
86 case BDSBeamPipeType::circular:
87 {return circular; break;}
88 case BDSBeamPipeType::elliptical:
89 {return elliptical; break;}
90 case BDSBeamPipeType::rectangular:
91 {return rectangular; break;}
92 case BDSBeamPipeType::lhc:
93 {return lhc; break;}
94 case BDSBeamPipeType::lhcdetailed:
95 {return lhcdetailed; break;}
96 case BDSBeamPipeType::rectellipse:
97 {return rectellipse; break;}
98 case BDSBeamPipeType::racetrack:
99 {return racetrack; break;}
100 case BDSBeamPipeType::octagonal:
101 {return octagonal; break;}
102 case BDSBeamPipeType::circularvacuum:
103 {return circularvacuum; break;}
104 case BDSBeamPipeType::clicpcl:
105 {return clicpcl; break;}
106 case BDSBeamPipeType::pointsfile:
107 {return pointsfile; break;}
108 case BDSBeamPipeType::rhombus:
109 {return rhombus; break;}
110 default:
111#ifdef BDSDEBUG
112 G4cout << __METHOD_NAME__ << "unknown type \"" << type << "\" - circular beampipe factory by default" << G4endl;
113#endif
114 return circular;
115 break;
116 }
117}
