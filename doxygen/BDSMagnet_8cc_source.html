50BDSMagnet::BDSMagnet(BDSMagnetType typeIn,
51 const G4String& nameIn,
52 G4double lengthIn,
53 BDSBeamPipeInfo* beamPipeInfoIn,
54 BDSMagnetOuterInfo* magnetOuterInfoIn,
55 BDSFieldInfo* vacuumFieldInfoIn,
56 G4double angleIn,
57 BDSFieldInfo* outerFieldInfoIn,
58 G4bool isThinIn):
59 BDSAcceleratorComponent(nameIn, lengthIn, angleIn, typeIn.ToString(), beamPipeInfoIn),
60 magnetType(typeIn),
61 magnetOuterInfo(magnetOuterInfoIn),
62 vacuumFieldInfo(vacuumFieldInfoIn),
63 outerFieldInfo(outerFieldInfoIn),
64 beampipe(nullptr),
65 placeBeamPipe(false),
66 magnetOuterOffset(G4ThreeVector(0,0,0)),
67 outer(nullptr),
68 beamPipePlacementTransform(G4Transform3D()),
69 isThin(isThinIn)
70{
71 magnetOuterInfo->name += "_outer";
72 horizontalWidth = magnetOuterInfoIn->horizontalWidth;
73 containerRadius = 0.5*horizontalWidth;
74
75 beampipe = nullptr;
76 outer = nullptr;
77
78 placeBeamPipe = false;
79
80 // It's not possible to build advanced outer geometry for a very thin magnet.
81 if (lengthIn < 1e-4*CLHEP::m) // 100um minimum length for geometry
82 {magnetOuterInfo->geometryType = BDSMagnetGeometryType::none;}
83 // No beam pipe geometry for really short 'magnets'
84 if (lengthIn < 1e-6*CLHEP::m)
85 {
86 GetBeamPipeInfo()->beamPipeType = BDSBeamPipeType::circularvacuum;
87 isThin = true;
88 }
89}

91G4String BDSMagnet::DetermineScalingKey(BDSMagnetType typeIn)
92{
93 G4String result = "none";
94 switch (typeIn.underlying())
95 {
96 case BDSMagnetType::hkicker:
97 case BDSMagnetType::vkicker:
98 case BDSMagnetType::muonspoiler:
99 case BDSMagnetType::sectorbend:
100 case BDSMagnetType::rectangularbend:
101 {result = "field"; break;}
102 case BDSMagnetType::quadrupole:
103 {result = "k1"; break;}
104 case BDSMagnetType::sextupole:
105 {result = "k2"; break;}
106 case BDSMagnetType::octupole:
107 {result = "k3"; break;}
108 case BDSMagnetType::decapole:
109 {result = "k4"; break;}
110 default:
111 {break;} // leave as none without complaint
112 };
113
114 return result;
115}

137void BDSMagnet::Build()
138{
139 BuildUserLimits();
140 BuildBeampipe();
141 BuildVacuumField();
142 BuildOuter();
143 // Instead of BDSAcceleratorComponent::Build just call BuildContainerLogicalVolume
144 // to control user limits in the case where there is no container and we just inherit

147 BuildOuterField(); // must be done when the containerLV exists

151void BDSMagnet::BuildBeampipe()
152{
153 beampipe = BDSBeamPipeFactory::Instance()->CreateBeamPipe(name+"_bp",
154 chordLength - 2*lengthSafety,
155 beamPipeInfo);
156
157 beamPipePlacementTransform = beampipe->GetPlacementTransform().inverse();
158
159 RegisterDaughter(beampipe);
160 InheritExtents(beampipe);
161
162 SetAcceleratorVacuumLogicalVolume(beampipe->GetVacuumLogicalVolume());
163
165 SetInputFaceNormal(BDS::RotateToReferenceFrame(beampipe->InputFaceNormal(), angle));
166 SetOutputFaceNormal(BDS::RotateToReferenceFrame(beampipe->OutputFaceNormal(), -angle));
167}

169void BDSMagnet::BuildVacuumField()
170{
171 if (vacuumFieldInfo)
172 {
173 if (beamPipePlacementTransform != G4Transform3D::Identity)
174 {
175 G4Transform3D newFieldTransform = vacuumFieldInfo->Transform() * beamPipePlacementTransform;
176 vacuumFieldInfo->SetTransform(newFieldTransform);
177 }
178 // can use containerLV for field as we don't construct any geometry with thin elements.
179 if (isThin)
180 {
181 BDSFieldBuilder::Instance()->RegisterFieldForConstruction(vacuumFieldInfo,
182 beampipe->GetContainerLogicalVolume(),
183 true);
184 }
185 else
186 {
187 BDSFieldBuilder::Instance()->RegisterFieldForConstruction(vacuumFieldInfo,
188 beampipe->GetVolumesForField(),
189 true);
190 }
191 }

194void BDSMagnet::BuildOuter()
195{
196 G4double outerLength = chordLength - 2*lengthSafety;
197 // The outerFieldInfo is required only so we don't reuse geometry when we need a unique field on it.
198 outer = BDSMagnetOuterFactory::Instance()->CreateMagnetOuter(magnetType,
199 magnetOuterInfo,
200 outerLength,
201 chordLength,
202 beampipe);
203
204 if (outer)
205 {
206 // copy necessary bits out of BDSGeometryComponent that holds
207 // container information for whole magnet object provided by
208 // magnet outer factory.
209 BDSGeometryComponent* container = outer->GetMagnetContainer();
210 containerSolid = container->GetContainerSolid()->Clone();
211 G4ThreeVector contOffset = container->GetPlacementOffset();
212 // set the main offset of the whole magnet which is placed w.r.t. the
213 // zero coordinate of the container solid
214 SetPlacementOffset(contOffset);
215
216 RegisterDaughter(outer);
217 InheritExtents(container, contOffset); // update extents
218
219 // Only clear after extents etc have been used
220 outer->ClearMagnetContainer();
221
222 endPieceBefore = outer->EndPieceBefore();
223 endPieceAfter = outer->EndPieceAfter();
224
226 SetInputFaceNormal(BDS::RotateToReferenceFrame(outer->InputFaceNormal(), angle));
227 SetOutputFaceNormal(BDS::RotateToReferenceFrame(outer->OutputFaceNormal(), -angle));
228 }
229}

231void BDSMagnet::BuildOuterField()
232{
233 if (outer && outerFieldInfo)
234 {
235 // determine key for this specific magnet instance
236 G4String scalingKey = DetermineScalingKey(magnetType);
237
238 BDSMagnetStrength* scalingStrength = vacuumFieldInfo ? vacuumFieldInfo->MagnetStrength() : nullptr;
239 G4LogicalVolume* vol = outer->GetContainerLogicalVolume();
240 BDSFieldBuilder::Instance()->RegisterFieldForConstruction(outerFieldInfo,
241 vol,
242 true,
243 scalingStrength,
244 scalingKey);
245 // Attach to the container but don't propagate to daughter volumes. This ensures
246 // any gap between the beam pipe and the outer also has a field.
247 BDSFieldBuilder::Instance()->RegisterFieldForConstruction(outerFieldInfo,
248 containerLogicalVolume,
249 false,
250 scalingStrength,
251 scalingKey);
252
253 // in the case of LHC-style geometry, override the second beam pipe (which is a daughter of the outer)
254 // field to be the opposite sign of the main vacuum field (same strength)
255 auto mgt = magnetOuterInfo->geometryType;
256 if (mgt == BDSMagnetGeometryType::lhcleft || mgt == BDSMagnetGeometryType::lhcright)
257 {
258 std::set<BDSGeometryComponent*> daughtersSet = outer->GetAllDaughters();
259 std::vector<BDSGeometryComponent*> daughters(daughtersSet.begin(), daughtersSet.end());
260 if (daughters.size() == 1 && vacuumFieldInfo)
261 {
262 BDSFieldInfo* secondBPField = new BDSFieldInfo(*vacuumFieldInfo);
263 G4double sign = mgt == BDSMagnetGeometryType::lhcleft ? 1.0 : -1.0;
264 secondBPField->Translate(G4ThreeVector(sign * BDSMagnetOuterFactoryLHC::beamSeparation, 0, 0));
265
266 if (magnetType != BDSMagnetType::quadrupole)
267 {
268 (*(secondBPField->MagnetStrength()))["field"] *= -1; // flip the sign
269 if (BDS::IsFinite((*(secondBPField->MagnetStrength()))["k1"]))
270 {(*(secondBPField->MagnetStrength()))["k1"] *= -1;}
271 }
272
273 secondBPField->SetIntegratorType(BDSIntegratorType::g4classicalrk4);
274 BDSFieldBuilder::Instance()->RegisterFieldForConstruction(secondBPField,
275 daughters[0]->GetContainerLogicalVolume(),
276 true);
277 }
278 }
279 }

282void BDSMagnet::BuildContainerLogicalVolume()
283{
284 // note beam pipe is not optional!
285 if (outer)
286 {//build around that
287 // container solid will have been updated in BuildOuter if the outer exists
288 containerLogicalVolume = new G4LogicalVolume(containerSolid,
289 emptyMaterial,
290 name + "_container_lv");
291
292 // user limits - provided by BDSAcceleratorComponent
293 containerLogicalVolume->SetUserLimits(userLimits);
294 containerLogicalVolume->SetVisAttributes(containerVisAttr);
295
296 placeBeamPipe = true;
297 }
298 else
299 {
300 // use beam pipe container volume as ours and no need to place beam pipe
301 containerSolid = beampipe->GetContainerSolid();
302 containerLogicalVolume = beampipe->GetContainerLogicalVolume();
303 InheritExtents(beampipe);
304 placeBeamPipe = false;
305 }
306}

308void BDSMagnet::PlaceComponents()
309{
310 if (placeBeamPipe)
311 {
312 G4ThreeVector beamPipeOffset = -1*GetPlacementOffset();
313 // place beampipe
314 G4PVPlacement* beamPipePV = new G4PVPlacement(nullptr, // rotation
315 beamPipeOffset, // position in container
316 beampipe->GetContainerLogicalVolume(), // its logical volume
317 name + "_beampipe_pv", // its name
318 containerLogicalVolume, // its mother volume
319 false, // no boolean operation
320 0, // copy number
321 checkOverlaps);
322
323 RegisterPhysicalVolume(beamPipePV);
324 }
325
326 if (outer)
327 {
328 //G4ThreeVector placementOffset = magnetOuterOffset + outer->GetPlacementOffset();
329 G4ThreeVector outerOffset = outer->GetPlacementOffset();
330
331 // place outer volume
332 G4PVPlacement* magnetOuterPV = new G4PVPlacement(nullptr, // rotation
333 outerOffset, // at normally (0,0,0)
334 outer->GetContainerLogicalVolume(), // its logical volume
335 name+"_outer_pv", // its name
336 containerLogicalVolume, // its mother volume
337 false, // no boolean operation
338 0, // copy number
339 checkOverlaps);
340
341 RegisterPhysicalVolume(magnetOuterPV);
342 }
343}
