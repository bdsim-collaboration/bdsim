38 // helper method
39 // replace algorithm of all substring instances
40 // from http://stackoverflow.com/questions/2896600/how-to-replace-all-occurrences-of-a-character-in-string
41 void replaceAll(std::string& source, const std::string& from, const std::string& to)
42 {
43 std::string newString;
44 newString.reserve( source.length() ); // avoids a few memory allocations
45
46 std::string::size_type lastPos = 0;
47 std::string::size_type findPos;
48
49 while( std::string::npos != ( findPos = source.find( from, lastPos )))
50 {
51 newString.append( source, lastPos, findPos - lastPos );
52 newString += to;
53 lastPos = findPos + from.length();
54 }
55
56 // Care for the rest after last occurrence
57 newString += source.substr( lastPos );
58
59 source.swap( newString );
60 }
61}

129Parser::Parser(std::string name)
130{
131 instance = this;
132#ifdef BDSDEBUG
133 std::cout << "gmad_parser> opening file" << std::endl;
134#endif
135 // replace all ~ symbols with home directory to allow for that
136 // note $HOME is not necessarily equivalent to ~
137 // see http://linux.die.net/man/3/getpwuid
138 std::string tilde("~");
139 std::string home(getpwuid(getuid())->pw_dir);
140
141 replaceAll(name,tilde,home);
142
143 FILE *f = fopen(name.c_str(),"r");
144
145 if(f==nullptr)
146 {
147 std::cerr << "gmad_parser> Can't open input file " << name << std::endl;
148 exit(1);
149 }
150 // set global string for parser
151 yyfilename = std::string(name);
152
153 Initialise();
154
155 std::cout.precision(10); // set output precision to 10 decimals
156
157 ParseFile(f);
158}

160void Parser::ParseFile(FILE *f)
161{
162 yyin=f;
163
164#ifdef BDSDEBUG
165 std::cout << "gmad_parser> beginning to parse file" << std::endl;
166#endif
167
168 while(!feof(yyin))
169 {yyparse();}
170
171 expand_sequences();
172#ifdef BDSDEBUG
173 std::cout << "gmad_parser> finished to parsing file" << std::endl;
174#endif
175 // clear temporary stuff
176#ifdef BDSDEBUG
177 std::cout << "gmad_parser> clearing temporary lists" << std::endl;
178#endif
179 element_list.clear();
180 tmp_list.clear();
181 symtab_map.clear();
182 for(auto it : var_list)
183 {delete it;}
184
185#ifdef BDSDEBUG
186 std::cout << "gmad_parser> finished" << std::endl;
187#endif
188
189 fclose(f);
190}

321void Parser::expand_line(FastList<Element>& target,
322 const std::string& name,
323 const std::string& start,
324 const std::string& end)
325{
326 const Element& line = find_element(name);
327 if(line.type != ElementType::_LINE && line.type != ElementType::_REV_LINE )
328 {
329 std::cerr << "Error with use command: \"" << name << "\" is not a line" << std::endl;
330 exit(1);
331 }
332
333 // delete the previous beamline
334 target.clear();
335
336 // expand the desired beamline
337 Element e;
338 e.type = line.type;
339 e.name = name;
340 e.l = 0;
341 e.lst = nullptr;
342
343 target.push_back(e);
344
345#ifdef BDSDEBUG
346 std::cout << "expanding line " << name << ", range = " << start << end << std::endl;
347#endif
348 if (!line.lst)
349 {return;} //list empty
350
351 // first expand the whole range
352 std::list<Element>::iterator sit = line.lst->begin();
353 std::list<Element>::iterator eit = line.lst->end();
354
355 // copy the list into the resulting list
356 switch(line.type)
357 {
358 case ElementType::_LINE:
359 {target.insert(target.end(),sit,eit); break;}
360 case ElementType::_REV_LINE:
361 {target.insert(target.end(),line.lst->rbegin(),line.lst->rend()); break;}
362 default:
363 {target.insert(target.end(),sit,eit); break;}
364 }
365 // bool to check if beamline is fully expanded
366 bool is_expanded = false;
367
368 // parse starting from the second element until the list is expanded
369 int iteration = 0;
370 while (!is_expanded)
371 {
372 is_expanded = true;
373 // start at second element
374 std::list<Element>::iterator it = ++target.begin();
375 for (; it!=target.end(); ++it)
376 {
377 Element& element = *it; // alias
378 const ElementType& type = element.type;
379#ifdef BDSDEBUG
380 std::cout << element.name << " , " << type << std::endl;
381#endif
382 // if list - expand further
383 if (type != ElementType::_LINE && type != ElementType::_REV_LINE)
384 {continue;}
385 is_expanded = false;
386 // lookup the line in main list
387 std::list<Element>::const_iterator tmpit = element_list.find(element.name);
388 std::list<Element>::const_iterator iterEnd = element_list.end();
389 if ( (tmpit != iterEnd) && ( (*tmpit).lst != nullptr) )
390 { // sublist found and not empty
391 const Element& list = *tmpit; // alias
392#ifdef BDSDEBUG
393 std::cout << "inserting sequence for " << element.name << " - " << list.name << " ...";
394#endif
395 if (type == ElementType::_LINE)
396 {target.insert(it,list.lst->begin(),list.lst->end());}
397 else if (type == ElementType::_REV_LINE)
398 {
399 //iterate over list and invert any sublines contained within. SPM
400 std::list<Element> tmpList;
401 tmpList.insert(tmpList.end(),list.lst->begin(),list.lst->end());
402 for (std::list<Element>::iterator itLineInverter = tmpList.begin();
403 itLineInverter != tmpList.end(); ++itLineInverter)
404 {
405 if ( (*itLineInverter).type == ElementType::_LINE)
406 {(*itLineInverter).type = ElementType::_REV_LINE;}
407 else if ((*itLineInverter).type == ElementType::_REV_LINE)
408 {(*itLineInverter).type = ElementType::_LINE;}
409 }
410 target.insert(it,tmpList.rbegin(),tmpList.rend());
411 }
412#ifdef BDSDEBUG
413 std::cout << "inserted" << std::endl;
414#endif
415 // delete the list pointer
416 target.erase(it--);
417 }
418 else if ( tmpit != iterEnd )
419 { // entry points to a scalar element type -
420 //transfer properties from the main list
421#ifdef BDSDEBUG
422 std::cout << "keeping element..." << element.name << std::endl;
423#endif
424 // copy properties
425 element = (*tmpit);
426
427#ifdef BDSDEBUG
428 std::cout << "done" << std::endl;
429#endif
430 }
431 else
432 { // element of undefined type
433 std::cerr << "Error : Expanding line \"" << name << "\" : element \"" << element.name
434 << "\" has not been defined! " << std::endl;
435 exit(1);
436 }
437 }
438 iteration++;
439 if ( iteration > MAX_EXPAND_ITERATIONS )
440 {
441 std::cerr << "Error : Line expansion of '" << name << "' seems to loop, " << std::endl
442 << "possible recursive line definition, quitting" << std::endl;
443 exit(1);
444 }
445 }// while
446
447 // leave only the desired range
448 //
449 // rule - from first occurrence of 'start' till first 'end' coming after 'start'
450
451 if ( !start.empty()) // determine the start element
452 {
453 std::list<Element>::const_iterator startIt = target.find(std::string(start));
454
455 if(startIt!=target.end())
456 {target.erase(target.begin(),startIt);}
457 }
458
459 if ( !end.empty()) // determine the end element
460 {
461 std::list<Element>::const_iterator endIt = target.find(std::string(end));
462
463 if(endIt!=target.end())
464 {target.erase(++endIt,target.end());}
465 }
466
467 // insert the tunnel if present
468
469 std::list<Element>::iterator itTunnel = element_list.find("tunnel");
470 if (itTunnel!=element_list.end())
471 {target.push_back(*itTunnel);}
472}

484void Parser::set_sampler(const std::string& name,
485 int count,
486 ElementType type,
487 const std::string& samplerType,
488 double samplerRadius,
489 int particleSetID)
490{
491 // if count equal to -2 add to all elements regardless of name
492 // typically used for output elements like samplers
493 // skip first element and add one at the end
494 if (count == -2)
495 {
496 for (auto it = beamline_list.begin(); it != beamline_list.end(); ++it)
497 {// skip LINEs
498 if((*it).type == ElementType::_LINE || (*it).type == ElementType::_REV_LINE)
499 {continue;}
500 // if type not equal to NONE and elements have to match type
501 if (type != ElementType::_NONE && type != (*it).type)
502 {continue;}
503
504 (*it).setSamplerInfo(samplerType,(*it).name,samplerRadius,particleSetID);
505 }
506 }
507 else if (count == -1) // if count equal to -1 add sampler to all element instances
508 {
509 auto itPair = beamline_list.equal_range(name);
510 if (itPair.first == itPair.second)
511 {
512 std::string msg = "parser> SetSampler> current beamline doesn't contain element \"" + name + "\"";
513 yyerror2(msg.c_str());
514 }
515 for (auto it = itPair.first; it != itPair.second; ++it)
516 {
517 // if sampler is attached to a marker, really attach it to the previous element with the name of marker
518 auto elementIt = (it->second);
519 std::string samplerName = elementIt->name;
520 if ((*elementIt).type == ElementType::_MARKER)
521 {
522 // need to find real element before
523 // but careful not to go beyond first element also!
524 while ((*elementIt).isSpecial())
525 {
526 elementIt--;
527 // have to break first before continue since in while loop
528 if (elementIt == beamline_list.begin())
529 {break;}
530 }
531
532 if (elementIt==beamline_list.begin())
533 {
534 std::cout << "parser> SetSampler> WARNING: no element before marker " << name << ", no sampler added" << std::endl;
535 continue;
536 }
537 }
538 (*elementIt).setSamplerInfo(samplerType,samplerName,samplerRadius,particleSetID);
539 }
540 }
541 else
542 {
543 auto it = beamline_list.find(name,count);
544 if (it==beamline_list.end())
545 {
546 std::string msg = "parser> SetSampler> current beamline doesn't contain element \"" + name + "\" with number " + std::to_string(count);
547 yyerror2(msg.c_str());
548 }
549 // if sampler is attached to a marker, really attach it to the previous element with the name of marker
550 std::string samplerName = (*it).name;
551 if ((*it).type == ElementType::_MARKER)
552 {
553 // need to find real element before
554 // but careful not to go beyond first element also!
555 while ((*it).isSpecial())
556 {
557 it--;
558 if (it == beamline_list.begin())
559 {
560 std::cout << "parser> SetSampler> WARNING: no element before marker " << name << ", no sampler added" << std::endl;
561 return;
562 }
563 }
564 }
565 (*it).setSamplerInfo(samplerType,samplerName,samplerRadius,particleSetID);
566 }
567}

751void Parser::Overwrite(const std::string& objectName)
752{
753 // find object and set values
754
755 // possible object types are:
756 // element, atom, colour, crystal, coolingchannel, field, laser, material, physicsbiasing, placement,
757 // query, region, tunnel, cavitymodel, samplerplacement, aperture, scorer, scorermesh, blm
758 bool extended = false;
759 auto element_it = element_list.find(objectName);
760 if (element_it != element_list.end())
761 {
762 ExtendObject(*element_it);
763 extended = true;
764 }
765 else
766 {
767 auto it = xsecbias_list.find(objectName);
768 if (it != xsecbias_list.end() )
769 {
770 ExtendObject(*it);
771 extended = true;
772 }
773 }
774 // vectors
775 if (!extended) {
776 if ( (extended = FindAndExtend<Atom> (objectName)) ) {}
777 else if ( (extended = FindAndExtend<NewColour> (objectName)) ) {}
778 else if ( (extended = FindAndExtend<Crystal> (objectName)) ) {}
779 else if ( (extended = FindAndExtend<CoolingChannel> (objectName)) ) {}
780 else if ( (extended = FindAndExtend<Field> (objectName)) ) {}
781 else if ( (extended = FindAndExtend<Material> (objectName)) ) {}
782 else if ( (extended = FindAndExtend<Placement> (objectName)) ) {}
783 else if ( (extended = FindAndExtend<Query> (objectName)) ) {}
784 else if ( (extended = FindAndExtend<Region> (objectName)) ) {}
785 else if ( (extended = FindAndExtend<Tunnel> (objectName)) ) {}
786 else if ( (extended = FindAndExtend<CavityModel>(objectName)) ) {}
787 else if ( (extended = FindAndExtend<SamplerPlacement>(objectName)) ) {}
788 else if ( (extended = FindAndExtend<Scorer> (objectName)) ) {}
789 else if ( (extended = FindAndExtend<ScorerMesh> (objectName)) ) {}
790 else if ( (extended = FindAndExtend<Aperture> (objectName)) ) {}
791 else if ( (extended = FindAndExtend<BLMPlacement> (objectName)) ) {}
792 else if ( (extended = FindAndExtend<Modulator> (objectName)) ) {}
793 else if ( (extended = FindAndExtend<Laser> (objectName)) ) {}
794 }
795
796 if (!extended)
797 {
798 std::cerr << "parser.h> Error: object \"" << objectName
799 << "\" has not been defined and can't be extended." << std::endl;
800 exit(1);
801 }
802
803 // clear maps
804 extendedNumbers.clear();
805 extendedStrings.clear();
806 extendedVectors.clear();
807}
