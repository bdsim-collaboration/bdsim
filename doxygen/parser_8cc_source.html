38 // helper method
39 // replace algorithm of all substring instances
40 // from http://stackoverflow.com/questions/2896600/how-to-replace-all-occurrences-of-a-character-in-string
41 void replaceAll(std::string& source, const std::string& from, const std::string& to)
42 {
43 std::string newString;
44 newString.reserve( source.length() ); // avoids a few memory allocations
45
46 std::string::size_type lastPos = 0;
47 std::string::size_type findPos;
48
49 while( std::string::npos != ( findPos = source.find( from, lastPos )))
50 {
51 newString.append( source, lastPos, findPos - lastPos );
52 newString += to;
53 lastPos = findPos + from.length();
54 }
55
56 // Care for the rest after last occurrence
57 newString += source.substr( lastPos );
58
59 source.swap( newString );
60 }
61}

358void Parser::expand_line(FastList<Element>& target,
359 const std::string& name,
360 const std::string& start,
361 const std::string& end)
362{
363 (*call_sequence_log) << "Parser::expand_line target=" << target.size() << " name=" << name << " start="
364 << start << " end=" << end << std::endl;
365
366 const Element& line = find_element(name);
367 if(line.type != ElementType::_LINE && line.type != ElementType::_REV_LINE )
368 {
369 std::cerr << "Error with use command: \"" << name << "\" is not a line" << std::endl;
370 exit(1);
371 }
372
373 // delete the previous beamline
374 target.clear();
375
376 // expand the desired beamline
377 Element e;
378 e.type = line.type;
379 e.name = name;
380 e.l = 0;
381 e.lst = nullptr;
382
383 target.push_back(e);
384
385#ifdef BDSDEBUG
386 std::cout << "expanding line " << name << ", range = " << start << end << std::endl;
387#endif
388 if (!line.lst)
389 {return;} //list empty
390
391 // first expand the whole range
392 std::list<Element>::iterator sit = line.lst->begin();
393 std::list<Element>::iterator eit = line.lst->end();
394
395 // copy the list into the resulting list
396 switch(line.type)
397 {
398 case ElementType::_LINE:
399 {target.insert(target.end(),sit,eit); break;}
400 case ElementType::_REV_LINE:
401 {target.insert(target.end(),line.lst->rbegin(),line.lst->rend()); break;}
402 default:
403 {target.insert(target.end(),sit,eit); break;}
404 }
405 // bool to check if beamline is fully expanded
406 bool is_expanded = false;
407
408 // parse starting from the second element until the list is expanded
409 int iteration = 0;
410 while (!is_expanded)
411 {
412 is_expanded = true;
413 // start at second element
414 std::list<Element>::iterator it = ++target.begin();
415 for (; it!=target.end(); ++it)
416 {
417 Element& element = *it; // alias
418 const ElementType& type = element.type;
419#ifdef BDSDEBUG
420 std::cout << element.name << " , " << type << std::endl;
421#endif
422 // if list - expand further
423 if (type != ElementType::_LINE && type != ElementType::_REV_LINE)
424 {continue;}
425 is_expanded = false;
426 // lookup the line in main list
427 std::list<Element>::const_iterator tmpit = element_list.find(element.name);
428 std::list<Element>::const_iterator iterEnd = element_list.end();
429 if ( (tmpit != iterEnd) && ( (*tmpit).lst != nullptr) )
430 { // sublist found and not empty
431 const Element& list = *tmpit; // alias
432#ifdef BDSDEBUG
433 std::cout << "inserting sequence for " << element.name << " - " << list.name << " ...";
434#endif
435 if (type == ElementType::_LINE)
436 {target.insert(it,list.lst->begin(),list.lst->end());}
437 else if (type == ElementType::_REV_LINE)
438 {
439 //iterate over list and invert any sublines contained within. SPM
440 std::list<Element> tmpList;
441 tmpList.insert(tmpList.end(),list.lst->begin(),list.lst->end());
442 for (std::list<Element>::iterator itLineInverter = tmpList.begin();
443 itLineInverter != tmpList.end(); ++itLineInverter)
444 {
445 if ( (*itLineInverter).type == ElementType::_LINE)
446 {(*itLineInverter).type = ElementType::_REV_LINE;}
447 else if ((*itLineInverter).type == ElementType::_REV_LINE)
448 {(*itLineInverter).type = ElementType::_LINE;}
449 }
450 target.insert(it,tmpList.rbegin(),tmpList.rend());
451 }
452#ifdef BDSDEBUG
453 std::cout << "inserted" << std::endl;
454#endif
455 // delete the list pointer
456 target.erase(it--);
457 }
458 else if ( tmpit != iterEnd )
459 { // entry points to a scalar element type -
460 //transfer properties from the main list
461#ifdef BDSDEBUG
462 std::cout << "keeping element..." << element.name << std::endl;
463#endif
464 // copy properties
465 element = (*tmpit);
466
467#ifdef BDSDEBUG
468 std::cout << "done" << std::endl;
469#endif
470 }
471 else
472 { // element of undefined type
473 std::cerr << "Error : Expanding line \"" << name << "\" : element \"" << element.name
474 << "\" has not been defined! " << std::endl;
475 exit(1);
476 }
477 }
478 iteration++;
479 if ( iteration > MAX_EXPAND_ITERATIONS )
480 {
481 std::cerr << "Error : Line expansion of '" << name << "' seems to loop, " << std::endl
482 << "possible recursive line definition, quitting" << std::endl;
483 exit(1);
484 }
485 }// while
486
487 // leave only the desired range
488 //
489 // rule - from first occurrence of 'start' till first 'end' coming after 'start'
490
491 if ( !start.empty()) // determine the start element
492 {
493 std::list<Element>::const_iterator startIt = target.find(std::string(start));
494
495 if(startIt!=target.end())
496 {target.erase(target.begin(),startIt);}
497 }
498
499 if ( !end.empty()) // determine the end element
500 {
501 std::list<Element>::const_iterator endIt = target.find(std::string(end));
502
503 if(endIt!=target.end())
504 {target.erase(++endIt,target.end());}
505 }
506
507 // insert the tunnel if present
508
509 std::list<Element>::iterator itTunnel = element_list.find("tunnel");
510 if (itTunnel!=element_list.end())
511 {target.push_back(*itTunnel);}
512}

539void Parser::set_sampler(const std::string& name,
540 int count,
541 ElementType type,
542 const std::string& samplerType,
543 double samplerRadius,
544 int particleSetID)
545{
546 (*call_sequence_log) << "Parser::set_sampler name=" << name
547 << " count=" << count
548 << " ElementType=" << type
549 << " samplerType=" << samplerType
550 << " samplerRadius=" << samplerRadius
551 << " particleSetID=" << particleSetID << std::endl;
552
553 // if count equal to -2 add to all elements regardless of name
554 // typically used for output elements like samplers
555 // skip first element and add one at the end
556 if (count == -2)
557 {
558 for (auto it = beamline_list.begin(); it != beamline_list.end(); ++it)
559 {// skip LINEs
560 if((*it).type == ElementType::_LINE || (*it).type == ElementType::_REV_LINE)
561 {continue;}
562 // if type not equal to NONE and elements have to match type
563 if (type != ElementType::_NONE && type != (*it).type)
564 {continue;}
565
566 (*it).setSamplerInfo(samplerType,(*it).name,samplerRadius,particleSetID);
567 }
568 }
569 else if (count == -1) // if count equal to -1 add sampler to all element instances
570 {
571 auto itPair = beamline_list.equal_range(name);
572 if (itPair.first == itPair.second)
573 {
574 std::string msg = "parser> SetSampler> current beamline doesn't contain element \"" + name + "\"";
575 yyerror2(msg.c_str());
576 }
577 for (auto it = itPair.first; it != itPair.second; ++it)
578 {
579 // if sampler is attached to a marker, really attach it to the previous element with the name of marker
580 auto elementIt = (it->second);
581 std::string samplerName = elementIt->name;
582 if ((*elementIt).type == ElementType::_MARKER)
583 {
584 // need to find real element before
585 // but careful not to go beyond first element also!
586 while ((*elementIt).isSpecial())
587 {
588 elementIt--;
589 // have to break first before continue since in while loop
590 if (elementIt == beamline_list.begin())
591 {break;}
592 }
593
594 if (elementIt==beamline_list.begin())
595 {
596 std::cout << "parser> SetSampler> WARNING: no element before marker " << name << ", no sampler added" << std::endl;
597 continue;
598 }
599 }
600 (*elementIt).setSamplerInfo(samplerType,samplerName,samplerRadius,particleSetID);
601 }
602 }
603 else
604 {
605 auto it = beamline_list.find(name,count);
606 if (it==beamline_list.end())
607 {
608 std::string msg = "parser> SetSampler> current beamline doesn't contain element \"" + name + "\" with number " + std::to_string(count);
609 yyerror2(msg.c_str());
610 }
611 // if sampler is attached to a marker, really attach it to the previous element with the name of marker
612 std::string samplerName = (*it).name;
613 if ((*it).type == ElementType::_MARKER)
614 {
615 // need to find real element before
616 // but careful not to go beyond first element also!
617 while ((*it).isSpecial())
618 {
619 it--;
620 if (it == beamline_list.begin())
621 {
622 std::cout << "parser> SetSampler> WARNING: no element before marker " << name << ", no sampler added" << std::endl;
623 return;
624 }
625 }
626 }
627 (*it).setSamplerInfo(samplerType,samplerName,samplerRadius,particleSetID);
628 }
629}

881void Parser::Overwrite(const std::string& objectName)
882{
883 (*call_sequence_log) << "Parser::Overwrite objectName=" << objectName << std::endl;
884
885 // find object and set values
886
887 // possible object types are:
888 // element, atom, colour, crystal, coolingchannel, field, laser, material, physicsbiasing, placement,
889 // query, region, tunnel, cavitymodel, samplerplacement, aperture, scorer, scorermesh, blm
890 bool extended = false;
891 auto element_it = element_list.find(objectName);
892 if (element_it != element_list.end())
893 {
894 ExtendObject(*element_it);
895 extended = true;
896 }
897 else
898 {
899 auto it = xsecbias_list.find(objectName);
900 if (it != xsecbias_list.end() )
901 {
902 ExtendObject(*it);
903 extended = true;
904 }
905 }
906 // vectors
907 if (!extended) {
908 if ( (extended = FindAndExtend<Atom> (objectName)) ) {}
909 else if ( (extended = FindAndExtend<NewColour> (objectName)) ) {}
910 else if ( (extended = FindAndExtend<Crystal> (objectName)) ) {}
911 else if ( (extended = FindAndExtend<CoolingChannel> (objectName)) ) {}
912 else if ( (extended = FindAndExtend<Field> (objectName)) ) {}
913 else if ( (extended = FindAndExtend<Material> (objectName)) ) {}
914 else if ( (extended = FindAndExtend<Placement> (objectName)) ) {}
915 else if ( (extended = FindAndExtend<Query> (objectName)) ) {}
916 else if ( (extended = FindAndExtend<Region> (objectName)) ) {}
917 else if ( (extended = FindAndExtend<Tunnel> (objectName)) ) {}
918 else if ( (extended = FindAndExtend<CavityModel>(objectName)) ) {}
919 else if ( (extended = FindAndExtend<SamplerPlacement>(objectName)) ) {}
920 else if ( (extended = FindAndExtend<Scorer> (objectName)) ) {}
921 else if ( (extended = FindAndExtend<ScorerMesh> (objectName)) ) {}
922 else if ( (extended = FindAndExtend<Aperture> (objectName)) ) {}
923 else if ( (extended = FindAndExtend<BLMPlacement> (objectName)) ) {}
924 else if ( (extended = FindAndExtend<Modulator> (objectName)) ) {}
925 else if ( (extended = FindAndExtend<Laser> (objectName)) ) {}
926 }
927
928 if (!extended)
929 {
930 std::cerr << "parser.h> Error: object \"" << objectName
931 << "\" has not been defined and can't be extended." << std::endl;
932 exit(1);
933 }
934
935 // clear maps
936 extendedNumbers.clear();
937 extendedStrings.clear();
938 extendedVectors.clear();
939}
