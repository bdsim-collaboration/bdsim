125int BDSIMLink::Initialise(double minimumKineticEnergy,
126 bool protonsAndIonsOnly)
127{
128 minimumKineticEnergy *= CLHEP::GeV; // units
129 G4bool trackerDebug = false;
130
132 const BDSExecOptions* execOptions = new BDSExecOptions(argcCache,argvCache);
133 if (usualPrintOut)
134 {execOptions->Print();}
135 ignoreSIGINT = execOptions->IgnoreSIGINT(); // different sig catching for cmake
136 execOptions->PrintCopyright();
137#ifdef BDSDEBUG
138 G4cout << __METHOD_NAME__ << "DEBUG mode is on." << G4endl;
139#endif
140
142 parser = BDSParser::Instance(execOptions->InputFileName());
144 parser->AmalgamateOptions(execOptions->Options());
145 parser->AmalgamateBeam(execOptions->Beam(), execOptions->Options().recreate);
147 parser->CheckOptions();
148
150 BDSMaterials::Instance()->PrepareRequiredMaterials(execOptions->Options().verbose);
151
153 delete execOptions;
154
158 BDSGlobalConstants* globalConstants = BDSGlobalConstants::Instance();
159
161 BDSRandom::CreateRandomNumberGenerator(globalConstants->RandomEngine());
162 BDSRandom::SetSeed(); // set the seed from options
163
165 bdsOutput = BDSOutputFactory::CreateOutput(globalConstants->OutputFormat(),
166 globalConstants->OutputFileName());
167
169 if (!BDS::Geant4EnvironmentIsSet())
170 {
171 G4cerr << "No Geant4 environmental variables found - please source geant4.sh environment" << G4endl;
172 G4cout << "A common fault is the wrong Geant4 environment as compared to the one BDSIM was compiled with." << G4endl;
173 return 1;
174 }
175
178 runManager = new BDSLinkRunManager();
179
181 construction = new BDSLinkDetectorConstruction();
182
184 auto parallelWorldsRequiringPhysics = BDS::ConstructAndRegisterParallelWorlds(construction, true, false);
185
186 runManager->SetUserInitialization(construction);
187
188 // Set filters used in sensitive detectors that transfer particles back
189 BDSSDManager::Instance()->SetLinkMinimumEK(minimumKineticEnergy);
190 BDSSDManager::Instance()->SetLinkProtonsAndIonsOnly(protonsAndIonsOnly);
191
192#ifdef BDSDEBUG
193 G4cout << __METHOD_NAME__ << "> Constructing physics processes" << G4endl;
194#endif
195 G4String physicsListName = parser->GetOptions().physicsList;
196
197#if G4VERSION_NUMBER > 1049
198 // from 10.5 onwards they have a looping particle killer that warnings and kills particles
199 // deemed to be looping that are <100 MeV. This is unrelated to the primary energy so troublesome.
200 // set to the 'low' limits here ~10keV. This must be done before any physics is created as the
201 // parameters are copied into the transportation physics process for each particle and it's very
202 // hard to sift through and fix afterwards
203 G4PhysicsListHelper::GetPhysicsListHelper()->UseLowLooperThresholds();
204#endif
205 auto parallelWorldPhysics = BDS::ConstructParallelWorldPhysics(parallelWorldsRequiringPhysics);
206 G4int physicsVerbosity = BDSGlobalConstants::Instance()->PhysicsVerbosity();
207 G4VModularPhysicsList* physList;
208 if (userPhysicsList)
209 {
210 G4cout << "Using externally registered user defined physics list" << G4endl;
211 physList = userPhysicsList;
212 }
213 else
214 {physList = BDS::BuildPhysics(physicsListName, physicsVerbosity);}
215
216 BDS::RegisterSamplerPhysics(parallelWorldPhysics, physList);
217 // Construction of the physics lists defines the necessary particles and therefore
218 // we can calculate the beam rigidity for the particle the beam is designed w.r.t. This
219 // must happen before the geometry is constructed (which is called by
220 // runManager->Initialize()).
221 BDSParticleDefinition* designParticle = nullptr;
222 BDSParticleDefinition* beamParticle = nullptr;
223 G4bool beamDifferentFromDesignParticle = false;
224 BDS::ConstructDesignAndBeamParticle(BDSParser::Instance()->GetBeam(),
225 globalConstants->FFact(),
226 designParticle,
227 beamParticle,
228 beamDifferentFromDesignParticle);
229 if (usualPrintOut)
230 {
231 G4cout << "Design particle properties: " << G4endl << *designParticle;
232 if (beamDifferentFromDesignParticle)
233 {G4cout << "Beam particle properties: " << G4endl << *beamParticle;}
234 }
235 // update rigidity where needed
236 construction->SetDesignParticle(designParticle);
237 //BDSFieldFactory::SetDesignParticle(designParticle);
238
239 //auto biasPhysics = BDS::BuildAndAttachBiasWrapper(parser->GetBiasing());
240 //if (biasPhysics)//could be nullptr and can't be passed to geant4 like this
241 // {physList->RegisterPhysics(biasPhysics);}
242 runManager->SetUserInitialization(physList);
243
245 GMAD::Beam b;
246 b.distrType = "sixtracklink";
247 if (!bdsBunch)
248 {
249 bdsBunch = BDSBunchFactory::CreateBunch(beamParticle,
250 parser->GetBeam(),
251 globalConstants->BeamlineTransform(),
252 globalConstants->BeamlineS(),
253 globalConstants->GeneratePrimariesOnly());
254 }
255 else
256 {
257 bdsBunch->SetOptions(beamParticle, b, BDSBunchType::reference); // update particle definition only
258 }
261 delete beamParticle;
262
264 G4GeometryTolerance* theGeometryTolerance = G4GeometryTolerance::GetInstance();
265 if (usualPrintOut)
266 {
267 G4cout << __METHOD_NAME__ << "Geometry Tolerances: " << G4endl;
268 G4cout << __METHOD_NAME__ << std::setw(12) << "Surface: " << std::setw(7) << theGeometryTolerance->GetSurfaceTolerance() << " mm" << G4endl;
269 G4cout << __METHOD_NAME__ << std::setw(12) << "Angular: " << std::setw(7) << theGeometryTolerance->GetAngularTolerance() << " rad" << G4endl;
270 G4cout << __METHOD_NAME__ << std::setw(12) << "Radial: " << std::setw(7) << theGeometryTolerance->GetRadialTolerance() << " mm" << G4endl;
271 }
273 runAction = new BDSLinkRunAction();
274 BDSLinkEventAction* eventAction = new BDSLinkEventAction(bdsOutput, runAction, trackerDebug);
275 runManager->SetUserAction(eventAction);
276 runManager->SetUserAction(runAction);
277 G4int verboseSteppingEventStart = globalConstants->VerboseSteppingEventStart();
278 G4int verboseSteppingEventStop = BDS::VerboseEventStop(verboseSteppingEventStart,
279 globalConstants->VerboseSteppingEventContinueFor());
280 runManager->SetUserAction(new BDSLinkTrackingAction(globalConstants->Batch(),
281 eventAction,
282 verboseSteppingEventStart,
283 verboseSteppingEventStop,
284 globalConstants->VerboseSteppingPrimaryOnly(),
285 globalConstants->VerboseSteppingLevel()));
286 runManager->SetUserAction(new BDSLinkStackingAction(globalConstants, std::set<G4int>(), protonsAndIonsOnly, minimumKineticEnergy));
287
288 /*
289 // Only add steppingaction if it is actually used, so do check here (for performance reasons)
290 G4int verboseSteppingEventStart = globalConstants->VerboseSteppingEventStart();
291 G4int verboseSteppingEventStop = BDS::VerboseEventStop(verboseSteppingEventStart,
292 globalConstants->VerboseSteppingEventContinueFor());
293 if (globalConstants->VerboseSteppingBDSIM())
294 {
295 runManager->SetUserAction(new BDSSteppingAction(true,
296 verboseSteppingEventStart,
297 verboseSteppingEventStop));
298 }
299 */
300
301 auto primaryGeneratorAction = new BDSLinkPrimaryGeneratorAction(bdsBunch, &currentElementIndex, construction, trackerDebug);
302 construction->SetPrimaryGeneratorAction(primaryGeneratorAction);
303 runManager->SetUserAction(primaryGeneratorAction);
304 //BDSFieldFactory::SetPrimaryGeneratorAction(primaryGeneratorAction);
305
307 runManager->Initialize();
308
310 construction->BuildPhysicsBias();
311
312 if (usualPrintOut && BDSGlobalConstants::Instance()->PhysicsVerbose())
313 {
314 BDS::PrintPrimaryParticleProcesses(bdsBunch->ParticleDefinition()->Name());
315 BDS::PrintDefinedParticles();
316 }
317
321 runManager->SetVerboseLevel(std::min(globalConstants->VerboseRunLevel(), globalConstants->PhysicsVerbosity()));
322 G4EventManager::GetEventManager()->SetVerboseLevel(globalConstants->VerboseEventLevel());
323 G4EventManager::GetEventManager()->GetTrackingManager()->SetVerboseLevel(globalConstants->VerboseTrackingLevel());
324
326 G4bool bCloseGeometry = G4GeometryManager::GetInstance()->CloseGeometry();
327 if (!bCloseGeometry)
328 {
329 G4cerr << __METHOD_NAME__ << "error - geometry not closed." << G4endl;
330 return 1;
331 }
332
333 if (globalConstants->ExportGeometry())
334 {
335 BDSGeometryWriter geometrywriter;
336 geometrywriter.ExportGeometry(globalConstants->ExportType(),
337 globalConstants->ExportFileName());
338 }
339
340 const auto& nameInds = construction->NameToElementIndex();
341 nameToElementIndex.insert(nameInds.begin(), nameInds.end());
342
343 if (bdsOutput)
344 {
345 bdsOutput->InitialiseGeometryDependent();
346 bdsOutput->NewFile();
347 }
348
349 initialised = true;
350 return 0;
351}

353void BDSIMLink::BeamOn(int nGenerate)
354{
355 if (initialisationResult > 1 || !initialised)
356 {return;} // a mode where we don't do anything
357
358 G4cout.precision(10);
360 struct sigaction act;
361 act.sa_handler = &BDS::HandleAborts;
362 sigemptyset(&act.sa_mask);
363 act.sa_flags = 0;
364 if (!ignoreSIGINT)
365 {sigaction(SIGINT, &act, nullptr);}
366 sigaction(SIGABRT, &act, nullptr);
367 sigaction(SIGTERM, &act, nullptr);
368 sigaction(SIGSEGV, &act, nullptr);
369
371 try
372 {
373 if (!BDSGlobalConstants::Instance()->Batch()) // Interactive mode
374 {
375 BDSVisManager visManager = BDSVisManager(BDSGlobalConstants::Instance()->VisMacroFileName(),
376 BDSGlobalConstants::Instance()->Geant4MacroFileName());
377 visManager.StartSession(argcCache, argvCache);
378 }
379 else
380 {// batch mode
381 if (nGenerate < 0)
382 {runManager->BeamOn(BDSGlobalConstants::Instance()->NGenerate());}
383 else
384 {runManager->BeamOn(nGenerate);}
385 }
386 }
387 catch (const BDSException& exception)
388 {
389 // don't do this for now in case it's dangerous and we try tracking with open geometry
390 //G4GeometryManager::GetInstance()->OpenGeometry();
391 G4cout << exception.what() << G4endl;
392 exit(1);
393 }
394}

396BDSIMLink::~BDSIMLink()
397{
399 G4GeometryManager::GetInstance()->OpenGeometry();
400
401#ifdef BDSDEBUG
402 G4cout << __METHOD_NAME__ << "deleting..." << G4endl;
403#endif
404 if (bdsOutput)
405 {bdsOutput->CloseFile();}
406 delete bdsOutput;
407
408 try
409 {
410 // order important here because of singletons relying on each other
411 delete BDSBeamPipeFactory::Instance();
412 delete BDSCavityFactory::Instance();
413 delete BDSGeometryFactory::Instance();
414 delete BDSAcceleratorModel::Instance();
415 delete BDSTemporaryFiles::Instance();
416 delete BDSFieldFactory::Instance(); // this uses BDSGlobalConstants which uses BDSMaterials
417 delete BDSGlobalConstants::Instance();
418 delete BDSMaterials::Instance();
419
420 // instances not used in this file, but no other good location for deletion
421 if (initialisationResult < 2)
422 {
423 delete BDSColours::Instance();
424 delete BDSFieldLoader::Instance();
425 //delete BDSSDManager::Instance();
426 delete BDSSamplerRegistry::Instance();
427 BDSAperturePointsCache::Instance()->ClearCachedFiles();
428 }
429 }
430 catch (...)
431 {;} // ignore any exception as this is a destructor
432
433 delete runManager;
434 delete parser;
435
436 if (usualPrintOut)
437 {G4cout << __METHOD_NAME__ << "End of Run. Thank you for using BDSIM!" << G4endl;}
438}

449const BDSLinkComponent* BDSIMLink::GetLinkComponent(int linkID) const
450{
451 const BDSBeamline* bl = construction->LinkBeamline();
452 if (!bl)
453 {return nullptr;}
454 if (linkID > (int)bl->size())
455 {return nullptr;}
456 const auto rawAccComponent = bl->at(linkID)->GetAcceleratorComponent();
457 const auto linkComponent = dynamic_cast<const BDSLinkComponent*>(rawAccComponent);
458 return linkComponent;
