67G4bool BDSSDSamplerLink::ProcessHits(G4Step* aStep, G4TouchableHistory* /*readOutTH*/)
68{
69 // Do not store hit if the particle pre step point is not on the boundary
70 G4StepPoint* postStepPoint = aStep->GetPostStepPoint();
71 if (postStepPoint->GetStepStatus() != fGeomBoundary)
72 {return false;} // this step was not stored
73
74 G4Track* track = aStep->GetTrack();
75 const G4DynamicParticle* dp = track->GetDynamicParticle();
76 G4double charge = dp->GetCharge(); // dynamic effective charge
77 auto pd = dp->GetParticleDefinition();
78
79 // check against various filters
80 G4double ek = track->GetKineticEnergy();
81 if (ek < minimumEK)
82 {return false;}
83
84 G4int PDGtype = pd->GetPDGEncoding();
85 if (protonsAndIonsOnly)
86 {
87 if (!BDS::IsIon(dp) && PDGtype != 2212)
88 {return false;}
89 }
90
91 G4int trackID = track->GetTrackID(); // unique ID of track
92 G4int parentID = track->GetParentID(); // unique ID of track's mother
93 G4double T = track->GetGlobalTime(); // time since beginning of event
94 G4double energy = track->GetTotalEnergy(); // total track energy
95
96 const G4ThreeVector& pos = track->GetPosition(); // current particle position (global)
97 const G4ThreeVector& mom = track->GetMomentumDirection(); // current particle direction (global) (unit)
98 G4double weight = track->GetWeight(); // weighting
99 G4int nElectrons = dp->GetTotalOccupancy();
100 G4double mass = dp->GetMass();
101 G4double momentum = dp->GetTotalMomentum();
102
103 G4int z = pd->GetAtomicNumber();
104 G4int a = pd->GetAtomicMass();
105
106 // The copy number from the physical volume is used as our unique sampler ID
107 G4int samplerID = track->GetVolume()->GetCopyNo();
108 //G4cout << "samplerID " << samplerID << G4endl;
109
110 // Initialize variables for the local position and direction
111 G4ThreeVector localPosition;
112 G4ThreeVector localDirection;
113
114 // Get coordinate transform and prepare local coordinates
115 G4Transform3D globalToLocal = G4Transform3D::Identity;
116 G4ThreeVector globalToLocalOffset = G4ThreeVector();
117 G4bool noRotation = true;
118 if (registry)
119 {
120 noRotation = registry->NoRotation(samplerID);
121 globalToLocal = registry->TransformInverse(samplerID);
122 globalToLocalOffset = globalToLocal.getTranslation();
123 }
124 if (noRotation)
125 {
126 localPosition = pos + globalToLocalOffset;
127 localDirection = mom;
128 }
129 else if (globalToLocal != G4Transform3D::Identity)
130 {
131 // The global to local transform is defined in the registry.
132 // Cast 3 vector to 'point' to transform position (required to be explicit for * operator)
133 localPosition = globalToLocal * (HepGeom::Point3D<G4double>)pos;
134 // Now, if the sampler is infinitely thin, the local z should be 0, but it's finite.
135 // Account for this by purposively setting local z to be 0.
136 localPosition.setZ(0.0);
137 // Cast 3 vector to 3 vector to transform vector (required to be explicit for * operator)
138 localDirection = globalToLocal * (HepGeom::Vector3D<G4double>)mom;
139 }
140
141 BDSParticleCoordsFull coords(localPosition.x(),
142 localPosition.y(),
143 localPosition.z(),
144 localDirection.x(),
145 localDirection.y(),
146 localDirection.z(),
147 T,
148 localPosition.z(), // s = z here
149 energy,
150 weight);
151
152 BDSHitSamplerLink* smpHit = new BDSHitSamplerLink(samplerID,
153 coords,
154 momentum,
155 mass,
156 z,
157 a,
158 charge,
159 PDGtype,
160 parentID,
161 trackID,
162 nElectrons);
163
164 samplerLinkCollection->insert(smpHit);
165 return true; // the hit was stored
166}
