134 template <typename T>
135 void FastList<T>::insert_before(const std::string& name, const T& val) {
136 FastMapIteratorPair itPair = equal_range(name);
137 if (itPair.first==itPair.second) {
138 std::cerr<<"current list doesn't contain element "<< name << std::endl;
139 exit(1);
140 }
141 // first insert into list and only then into map, otherwise map iterators are invalidated(!)
142 std::vector<FastListIterator> listIterators;
143 for (FastMapIterator it = itPair.first; it != itPair.second; ++it)
144 {
145 FastListIterator listIt = itsList.insert(it->second,val);
146 listIterators.push_back(listIt);
147 }
148 for (FastListIterator it : listIterators) {
149 itsMap.insert(std::pair<std::string,FastListIterator>(val.name,it));
150 }
151 }

153 template <typename T>
154 void FastList<T>::push_back(const T& el, bool unique, const std::string& className) {
155 // better to search in map (faster)
156 if (unique && itsMap.find(el.name) != itsMap.end()) {
157 std::cout << "ERROR: " << className << " with name \"" << el.name << "\" already defined." << std::endl;
158 exit(1);
159 }
160 // insert at back of list (insert() instead of push_back() to get iterator for map):
161 FastListIterator it = itsList.insert(end(),el);
162 itsMap.insert(std::pair<std::string,FastListIterator>(el.name,it));
163 }

185 template <typename T>
186 typename FastList<T>::FastListConstIterator FastList<T>::erase(const FastList<T>::FastListConstIterator it) {
187
188 // find entry in map to erase:
189 std::string name = (*it).name;
190 if (itsMap.count(name) == 1) {
191 itsMap.erase(name);
192 }
193 else { // more than one entry with same name
194 FastMapIteratorPair ret = itsMap.equal_range(name);
195 for (FastMapIterator emit = ret.first; emit!=ret.second; ++emit) {
196 if ((*emit).second == it) // valid comparison? if not, how to find correct element?
197 {
198 itsMap.erase(emit);
199 break;
200 }
201 }
202 }
203 return itsList.erase(it);
204 }

255 template <typename T>
256 typename FastList<T>::FastListConstIterator FastList<T>::find(std::string name,unsigned int count)const {
257 if (count==1) {
258 FastMapConstIterator emit = itsMap.find(name);
259 if (emit==itsMap.end()) return itsList.end();
260 return (*emit).second;
261 } else {
262 // if count > 1
263 FastMapConstIteratorPair ret = itsMap.equal_range(name);
264 unsigned int i=1;
265 for (FastMapConstIterator emit = ret.first; emit!=ret.second; ++emit, i++) {
266 if (i==count) {
267 return (*emit).second;
268 }
269 }
270 return itsList.end();
271 }

276 if (count==1) {
277 FastMapIterator emit = itsMap.find(name);
278 if (emit==itsMap.end()) return itsList.end();
279 // remove constness trick, call erase with empty range
280 FastListIterator it = itsList.erase((*emit).second, (*emit).second);
281 return it;
282 } else {
283 // if count > 1
284 FastMapIteratorPair ret = itsMap.equal_range(name);
285 unsigned int i=1;
286 for (FastMapIterator emit = ret.first; emit!=ret.second; ++emit, i++) {
287 if (i==count) {
288 // remove constness trick, call erase with empty range
289 FastListIterator it = itsList.erase((*emit).second, (*emit).second);
290 return it;
291 }
292 }
293 return itsList.end();
