135 template <typename T>
136 void FastList<T>::insert_before(const std::string& name, const T& val) {
137 FastMapIteratorPair itPair = equal_range(name);
138 if (itPair.first==itPair.second) {
139 std::cerr<<"current list doesn't contain element "<< name << std::endl;
140 exit(1);
141 }
142 // first insert into list and only then into map, otherwise map iterators are invalidated(!)
143 std::vector<FastListIterator> listIterators;
144 for (FastMapIterator it = itPair.first; it != itPair.second; ++it)
145 {
146 FastListIterator listIt = itsList.insert(it->second,val);
147 listIterators.push_back(listIt);
148 }
149 for (FastListIterator it : listIterators) {
150 itsMap.insert(std::pair<std::string,FastListIterator>(val.name,it));
151 }
152 }

154 template <typename T>
155 void FastList<T>::push_back(const T& el, bool unique, const std::string& className) {
156 // better to search in map (faster)
157 if (unique && itsMap.find(el.name) != itsMap.end()) {
158 std::cout << "ERROR: " << className << " with name \"" << el.name << "\" already defined." << std::endl;
159 exit(1);
160 }
161 // insert at back of list (insert() instead of push_back() to get iterator for map):
162 FastListIterator it = itsList.insert(end(),el);
163 itsMap.insert(std::pair<std::string,FastListIterator>(el.name,it));
164 }

186 template <typename T>
187 typename FastList<T>::FastListConstIterator FastList<T>::erase(const FastList<T>::FastListConstIterator it) {
188
189 // find entry in map to erase:
190 std::string name = (*it).name;
191 if (itsMap.count(name) == 1) {
192 itsMap.erase(name);
193 }
194 else { // more than one entry with same name
195 FastMapIteratorPair ret = itsMap.equal_range(name);
196 for (FastMapIterator emit = ret.first; emit!=ret.second; ++emit) {
197 if ((*emit).second == it) // valid comparison? if not, how to find correct element?
198 {
199 itsMap.erase(emit);
200 break;
201 }
202 }
203 }
204 return itsList.erase(it);
205 }

256 template <typename T>
257 typename FastList<T>::FastListConstIterator FastList<T>::find(std::string name,unsigned int count)const {
258 if (count==1) {
259 FastMapConstIterator emit = itsMap.find(name);
260 if (emit==itsMap.end()) return itsList.end();
261 return (*emit).second;
262 } else {
263 // if count > 1
264 FastMapConstIteratorPair ret = itsMap.equal_range(name);
265 unsigned int i=1;
266 for (FastMapConstIterator emit = ret.first; emit!=ret.second; ++emit, i++) {
267 if (i==count) {
268 return (*emit).second;
269 }
270 }
271 return itsList.end();
272 }

277 if (count==1) {
278 FastMapIterator emit = itsMap.find(name);
279 if (emit==itsMap.end()) return itsList.end();
280 // remove constness trick, call erase with empty range
281 FastListIterator it = itsList.erase((*emit).second, (*emit).second);
282 return it;
283 } else {
284 // if count > 1
285 FastMapIteratorPair ret = itsMap.equal_range(name);
286 unsigned int i=1;
287 for (FastMapIterator emit = ret.first; emit!=ret.second; ++emit, i++) {
288 if (i==count) {
289 // remove constness trick, call erase with empty range
290 FastListIterator it = itsList.erase((*emit).second, (*emit).second);
291 return it;
292 }
293 }
