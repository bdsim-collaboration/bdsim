149int BDSIM::Initialise()
150{
151
152#ifdef USE_GEANT4CGALBOOLEAN
153 G4BooleanSolid::SetExternalBooleanProcessor(new G4BooleanProcessorCGAL());
154#endif
155
157 const BDSExecOptions* execOptions = new BDSExecOptions(argcCache,argvCache);
158 if (usualPrintOut)
159 {execOptions->Print();}
160 ignoreSIGINT = execOptions->IgnoreSIGINT(); // different sig catching for cmake
161
162 execOptions->PrintCopyright();
163#ifdef BDSDEBUG
164 G4cout << __METHOD_NAME__ << "DEBUG mode is on." << G4endl;
165#endif
166
168 if (!parser) {
169 parser = BDSParser::Instance(execOptions->InputFileName());
171 }
172
173 parser->AmalgamateOptions(execOptions->Options());
174 parser->AmalgamateBeam(execOptions->Beam(),
175 execOptions->Options().recreate);
177 parser->CheckOptions();
178
179
181 BDSMaterials::Instance()->PrepareRequiredMaterials(execOptions->Options().verbose);
182
184 delete execOptions;
185
189 BDSGlobalConstants* globals = BDSGlobalConstants::Instance();
190
192 BDSRandom::CreateRandomNumberGenerator(globals->RandomEngine());
193 BDSRandom::SetSeed(); // set the seed from options
194
196 bdsOutput = BDSOutputFactory::CreateOutput(globals->OutputFormat(),
197 globals->OutputFileName());
198
200 if (!BDS::Geant4EnvironmentIsSet())
201 {
202 G4cerr << "No Geant4 environmental variables found - please source geant4.sh environment" << G4endl;
203 G4cout << "A common fault is the wrong Geant4 environment as compared to the one BDSIM was compiled with." << G4endl;
204 return 1;
205 }
206
209 runManager = new BDSRunManager();
210
212 realWorld = new BDSDetectorConstruction(userComponentFactory);
213
215 auto parallelWorldsRequiringPhysics = BDS::ConstructAndRegisterParallelWorlds(realWorld,
216 realWorld->BuildSamplerWorld(),
217 realWorld->BuildPlacementFieldsWorld());
218 runManager->SetUserInitialization(realWorld);
219
222#ifdef BDSDEBUG
223 G4cout << __METHOD_NAME__ << "> Constructing physics processes" << G4endl;
224#endif
225 G4String physicsListName = parser->GetOptions().physicsList;
226
227#if G4VERSION_NUMBER > 1049
228 // from 10.5 onwards they have a looping particle killer that warnings and kills particles
229 // deemed to be looping that are <100 MeV. This is unrelated to the primary energy so troublesome.
230 // set to the 'low' limits here ~10keV. This must be done before any physics is created as the
231 // parameters are copied into the transportation physics process for each particle and it's very
232 // hard to sift through and fix afterwards
233 G4PhysicsListHelper::GetPhysicsListHelper()->UseLowLooperThresholds();
234#endif
235 // sampler physics process for parallel world tracking must be instantiated BEFORE
236 // regular physics.
237 // Note, we purposively don't create a parallel world process for the curvilinear
238 // world as we don't need the track information from it - unreliable that way. We
239 // query the geometry directly using our BDSAuxiliaryNavigator class.
240 auto parallelWorldPhysics = BDS::ConstructParallelWorldPhysics(parallelWorldsRequiringPhysics);
241 G4int physicsVerbosity = globals->PhysicsVerbosity();
242 G4VModularPhysicsList* physList;
243 if (userPhysicsList)
244 {
245 G4cout << "Using externally registered user defined physics list" << G4endl;
246 physList = userPhysicsList;
247 }
248 else
249 {physList = BDS::BuildPhysics(physicsListName, physicsVerbosity);}
250
251 // create geometry sampler and register importance sampling biasing. Has to be here
252 // before physicsList is "initialised" in run manager.
253 if (BDSGlobalConstants::Instance()->UseImportanceSampling())
254 {BDS::RegisterImportanceBiasing(parallelWorldsRequiringPhysics, physList);}
255 if (BDSGlobalConstants::Instance()->ExtendPionDecayChannels())
256 {BDS::ExtendPionDecayChannels(physList);}
257 if (BDSGlobalConstants::Instance()->TurnOffMuonDecay())
258 {BDS::TurnOffMuonDecay();}
259
260 // Construction of the physics lists defines the necessary particles and therefore
261 // we can calculate the beam rigidity for the particle the beam is designed w.r.t. This
262 // must happen before the geometry is constructed (which is called by
263 // runManager->Initialize()).
264 BDSParticleDefinition* designParticle = nullptr;
265 BDSParticleDefinition* beamParticle = nullptr;
266 G4bool beamDifferentFromDesignParticle = false;
267 BDS::ConstructDesignAndBeamParticle(BDSParser::Instance()->GetBeam(),
268 globals->FFact(),
269 designParticle,
270 beamParticle,
271 beamDifferentFromDesignParticle);
272 G4double minEK = globals->MinimumKineticEnergy();
273 if (beamParticle->KineticEnergy() < minEK && BDS::IsFinite(minEK))
274 {throw BDSException("option, minimumKineticEnergy is higher than kinetic energy of the beam - all primary particles wil be killed!");}
275 if (usualPrintOut)
276 {
277 G4cout << "Design particle properties: " << G4endl << *designParticle;
278 if (beamDifferentFromDesignParticle)
279 {G4cout << "Beam particle properties: " << G4endl << *beamParticle;}
280 }
281 // update rigidity where needed
282 realWorld->SetDesignParticle(designParticle);
283 BDSFieldFactory::SetDesignParticle(designParticle);
284 BDSGeometryFactorySQL::SetDefaultRigidity(designParticle->BRho()); // used for sql field loading
285
286 // Muon splitting - optional - should be done *after* biasing to work with it - TBC it's before...
287 BDS::BuildMuonBiasing(physList);
288
289 BDS::RegisterSamplerPhysics(parallelWorldPhysics, physList);
290 auto biasPhysics = BDS::BuildAndAttachBiasWrapper(parser->GetBiasing());
291 if (biasPhysics)//could be nullptr and can't be passed to geant4 like this
292 {physList->RegisterPhysics(biasPhysics);}
293 runManager->SetUserInitialization(physList);
294
296 bdsBunch = BDSBunchFactory::CreateBunch(beamParticle,
297 parser->GetBeam(),
298 globals->BeamlineTransform(),
299 globals->BeamlineS(),
300 globals->GeneratePrimariesOnly());
301 G4cout << "Bunch distribution: \"" << bdsBunch->Name() << "\"" << G4endl;
304 delete beamParticle;
306 if (bdsBunch->ExpectChangingParticleType())
307 {BDS::ConstructExtendedParticleSet();}
308
312 if (globals->GeneratePrimariesOnly())
313 {
314 GeneratePrimariesOnly(globals);
315 return 0;
316 }
317
319 G4GeometryTolerance* theGeometryTolerance = G4GeometryTolerance::GetInstance();
320 if (usualPrintOut)
321 {
322 G4cout << __METHOD_NAME__ << "Geometry Tolerances: " << G4endl;
323 G4cout << __METHOD_NAME__ << std::setw(12) << "Surface: " << std::setw(7) << theGeometryTolerance->GetSurfaceTolerance() << " mm" << G4endl;
324 G4cout << __METHOD_NAME__ << std::setw(12) << "Angular: " << std::setw(7) << theGeometryTolerance->GetAngularTolerance() << " rad" << G4endl;
325 G4cout << __METHOD_NAME__ << std::setw(12) << "Radial: " << std::setw(7) << theGeometryTolerance->GetRadialTolerance() << " mm" << G4endl;
326 }
327
329 BDSEventAction* eventAction = new BDSEventAction(bdsOutput);
330 runManager->SetUserAction(eventAction);
331
332 BDSRunAction* runAction = new BDSRunAction(bdsOutput,
333 bdsBunch,
334 bdsBunch->ParticleDefinition()->IsAnIon(),
335 eventAction,
336 globals->StoreTrajectorySamplerID());
337 runManager->SetUserAction(runAction);
338
339 // Only add stepping action if it is actually used, so do check here (for performance reasons)
340 G4int verboseSteppingEventStart = globals->VerboseSteppingEventStart();
341 G4int verboseSteppingEventStop = BDS::VerboseEventStop(verboseSteppingEventStart,
342 globals->VerboseSteppingEventContinueFor());
343 if (globals->VerboseSteppingBDSIM())
344 {
345 runManager->SetUserAction(new BDSSteppingAction(true,
346 verboseSteppingEventStart,
347 verboseSteppingEventStop));
348 }
349
350 runManager->SetUserAction(new BDSTrackingAction(globals->Batch(),
351 globals->StoreTrajectory(),
352 globals->StoreTrajectoryOptions(),
353 eventAction,
354 verboseSteppingEventStart,
355 verboseSteppingEventStop,
356 globals->VerboseSteppingPrimaryOnly(),
357 globals->VerboseSteppingLevel(),
358 new BDSPolarizationState(parser->GetBeam())));
359
360 runManager->SetUserAction(new BDSStackingAction(globals));
361
362 auto primaryGeneratorAction = new BDSPrimaryGeneratorAction(bdsBunch, parser->GetBeam(), globals->Batch());
363 // possibly updated after the primary generator as loaded a beam file
364 eventAction->SetPrintModulo(BDSGlobalConstants::Instance()->PrintModuloEvents());
365 runManager->SetUserAction(primaryGeneratorAction);
366 BDSFieldFactory::SetPrimaryGeneratorAction(primaryGeneratorAction);
367
369 runManager->Initialize();
370
372 if (globals->UseImportanceSampling())
373 {BDS::AddIStore(parallelWorldsRequiringPhysics);}
374
376 realWorld->BuildPhysicsBias();
377
378 if (usualPrintOut && globals->PhysicsVerbose())
379 {
380 BDS::PrintPrimaryParticleProcesses(bdsBunch->ParticleDefinition()->Name());
381 BDS::PrintDefinedParticles();
382 }
383
387 runManager->SetVerboseLevel(std::min(globals->VerboseRunLevel(), globals->PhysicsVerbosity()));
388 G4EventManager::GetEventManager()->SetVerboseLevel(globals->VerboseEventLevel());
389 G4EventManager::GetEventManager()->GetTrackingManager()->SetVerboseLevel(globals->VerboseTrackingLevel());
390
392 G4bool bCloseGeometry = G4GeometryManager::GetInstance()->CloseGeometry();
393 if (!bCloseGeometry)
394 {
395 G4cerr << __METHOD_NAME__ << "error - geometry not closed." << G4endl;
396 return 1;
397 }
398
399 if (globals->ExportGeometry())
400 {
401 BDSGeometryWriter geometrywriter;
402 geometrywriter.ExportGeometry(globals->ExportType(),
403 globals->ExportFileName());
404 }
405
406 initialised = true;
407 return 0;
408}

410void BDSIM::BeamOn(int nGenerate)
411{
412 if (initialisationResult > 1 || !initialised)
413 {return;} // a mode where we don't do anything
414
415 G4cout.precision(10);
417 struct sigaction act;
418 act.sa_handler = &BDS::HandleAborts;
419 sigemptyset(&act.sa_mask);
420 act.sa_flags = 0;
421 if (!ignoreSIGINT)
422 {sigaction(SIGINT, &act, nullptr);}
423 sigaction(SIGABRT, &act, nullptr);
424 sigaction(SIGTERM, &act, nullptr);
425 sigaction(SIGSEGV, &act, nullptr);
426
428 try
429 {
430 if (!BDSGlobalConstants::Instance()->Batch()) // Interactive mode
431 {
432 BDSVisManager visManager = BDSVisManager(BDSGlobalConstants::Instance()->VisMacroFileName(),
433 BDSGlobalConstants::Instance()->Geant4MacroFileName(),
434 realWorld,
435 BDSGlobalConstants::Instance()->VisVerbosity());
436 visManager.StartSession(argcCache, argvCache);
437 }
438 else
439 {// batch mode
440 if (nGenerate < 0)
441 {runManager->BeamOn(BDSGlobalConstants::Instance()->NGenerate());}
442 else
443 {runManager->BeamOn(nGenerate);}
444 }
445 }
446 catch (const BDSException& exception)
447 {
448 // don't do this for now in case it's dangerous and we try tracking with open geometry
449 //G4GeometryManager::GetInstance()->OpenGeometry();
450 throw exception;
451 }
452}

454BDSIM::~BDSIM()
455{
457 G4GeometryManager::GetInstance()->OpenGeometry();
458
459#ifdef BDSDEBUG
460 G4cout << __METHOD_NAME__ << "deleting..." << G4endl;
461#endif
462 delete bdsOutput;
463
464 try
465 {
466 // order important here because of singletons relying on each other
467 delete BDSSDManager::Instance();
468 delete BDSBeamPipeFactory::Instance();
469 delete BDSCavityFactory::Instance();
470 delete BDSGeometryFactory::Instance();
471 delete BDSAcceleratorModel::Instance();
472 delete BDSTemporaryFiles::Instance();
473 delete BDSFieldFactory::Instance(); // this uses BDSGlobalConstants which uses BDSMaterials
474 delete BDSGlobalConstants::Instance();
475 delete BDSMaterials::Instance();
476
477 // instances not used in this file, but no other good location for deletion
478 if (initialisationResult < 2)
479 {
480 delete BDSColours::Instance();
481 delete BDSFieldLoader::Instance();
482 delete BDSSamplerRegistry::Instance();
483 BDSAperturePointsCache::Instance()->ClearCachedFiles();
484 }
485 }
486 catch (...)
487 {;} // ignore any exception as this is a destructor
488
489 delete runManager;
490 delete bdsBunch;
491
492 if (usualPrintOut)
493 {G4cout << __METHOD_NAME__ << "End of Run. Thank you for using BDSIM!" << G4endl;}
494}

508void BDSIM::GeneratePrimariesOnly(const BDSGlobalConstants* globals)
509{
510 // output creation is duplicated below but with this if loop, we exit so ok.
511 bdsOutput->NewFile();
512 const G4int nToGenerate = globals->NGenerate();
513 const G4int printModulo = globals->PrintModuloEvents();
514 bdsBunch->BeginOfRunAction(nToGenerate, globals->Batch());
515 auto flagsCache(G4cout.flags());
516 for (G4int i = 0; i < nToGenerate; i++)
517 {
518 if (i%printModulo == 0)
519 {G4cout << "\r Primary> " << std::fixed << i << " of " << nToGenerate << G4endl;}
520 BDSParticleCoordsFullGlobal coords = bdsBunch->GetNextParticleValid();
521 // always pull particle definition in case it's updated
522 const BDSParticleDefinition* pDef = bdsBunch->ParticleDefinition();
523 bdsOutput->FillEventPrimaryOnly(coords, pDef);
524 }
525 G4cout.flags(flagsCache); // restore cout flags
526 // Write options now the file is open
527 const GMAD::OptionsBase* ob = BDSParser::Instance()->GetOptionsBase();
528 bdsOutput->FillOptions(ob);
529
530 // Write beam
531 const GMAD::BeamBase* bb = BDSParser::Instance()->GetBeamBase();
532 bdsOutput->FillBeam(bb);
533
534 bdsOutput->CloseFile();
535}
