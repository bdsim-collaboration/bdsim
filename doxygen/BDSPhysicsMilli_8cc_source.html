67void BDSPhysicsMilli::ConstructProcess()
68{
69 //G4CoulombScattering* hCoul = nullptr;
70 //hCoul = new G4CoulombScattering();
71 auto* hMpl = new G4hMultipleScattering();
72 G4AutoDelete::Register(hMpl);
73 auto* modelmpl = new G4WentzelVIModel();
74 G4AutoDelete::Register(modelmpl);
75 modelmpl->SetActivationLowEnergyLimit(1.0*CLHEP::MeV);
76 hMpl->SetEmModel(modelmpl, 1);
77
78 G4PhysicsListHelper* ph = G4PhysicsListHelper::GetPhysicsListHelper();
79
80#if G4VERSION_NUMBER > 1029
81 auto aParticleIterator = GetParticleIterator();
82#endif
83 aParticleIterator->reset();
84
85 while( (*aParticleIterator)() )
86 {
87 G4ParticleDefinition* particle = aParticleIterator->value();
88 G4String particleName = particle->GetParticleName();
89
90 if (particleName == millichargeName)
91 {
92 //ph->RegisterProcess(hCoul, particle);
93 ph->RegisterProcess(hMpl, particle);
94 ph->RegisterProcess(new G4StepLimiter(), particle);
95 continue;
96 }
97 }
98}
