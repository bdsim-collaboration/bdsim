82BDSParticleCoordsFull BDSBunchBox::GetNextParticleLocal()
83{
84 G4double x = X0 + envelopeX * (1-2*G4RandFlat::shoot());
85 G4double y = Y0 + envelopeY * (1-2*G4RandFlat::shoot());
86 G4double z = Z0 + envelopeZ * (1-2*G4RandFlat::shoot());
87 G4double xp = envelopeXp * (1-2*G4RandFlat::shoot());
88 G4double yp = envelopeYp * (1-2*G4RandFlat::shoot());
89 G4double zp = envelopeZp * (1-2*G4RandFlat::shoot());
90 G4ThreeVector dir(xp,yp,zp);
91 dir = dir.unit();
92 xp = dir.x();
93 yp = dir.y();
94 zp = dir.z();
95
96 G4double t = T0 + envelopeT * (1.- 2.*G4RandFlat::shoot());
97 G4double E = E0 + envelopeE * (1 - 2.*G4RandFlat::shoot());
98
99 return BDSParticleCoordsFull(x,y,z,xp,yp,zp,t,S0+z,E,/*weight=*/1.0);
100}
