120BDSExtent BDSExtent::Tilted(G4double angle) const
121{
122 if (!BDS::IsFinite(angle))
123 {return BDSExtent(*this);}
124
125 // rotate each vector (from origin to each corner) by angle
126 // and check - safer than checking based on +ve / -ve angle
127 G4TwoVector topRight = G4TwoVector(extXPos, extYPos);
128 G4TwoVector botRight = G4TwoVector(extXPos, extYNeg);
129 G4TwoVector botLeft = G4TwoVector(extXNeg, extYNeg);
130 G4TwoVector topLeft = G4TwoVector(extXNeg, extYPos);
131
132 topRight.rotate(angle);
133 botRight.rotate(angle);
134 botLeft.rotate(angle);
135 topLeft.rotate(angle);
136
137 G4double xMin = std::min({botRight.x(), botLeft.x(), topLeft.x(), topRight.x()});
138 G4double xMax = std::max({botRight.x(), botLeft.x(), topLeft.x(), topRight.x()});
139 G4double yMin = std::min({botRight.y(), botLeft.y(), topLeft.y(), topRight.y()});
140 G4double yMax = std::max({botRight.y(), botLeft.y(), topLeft.y(), topRight.y()});
141
142 BDSExtent result = BDSExtent(xMin, xMax,
143 yMin, yMax,
144 extZNeg, extZPos);
145 return result;
146}

189G4bool BDSExtent::Encompasses(const G4ThreeVector& point) const
190{
191 G4bool xOK = (point.x() >= extXNeg) && (point.x() <= extXPos);
192 G4bool yOK = (point.y() >= extYNeg) && (point.y() <= extYPos);
193 G4bool zOK = (point.z() >= extZNeg) && (point.z() <= extZPos);
194 return xOK && yOK && zOK;
195}

197G4bool BDSExtent::Encompasses(const BDSExtent& other) const
198{
199 std::vector<G4ThreeVector> otherPoints = other.AllBoundaryPoints();
200 G4bool result = false;
201 for (const auto& p : otherPoints)
202 {result = result || !Encompasses(p);}
203 return !result;
204}

213{
214 margin = std::abs(margin);

235void BDSExtent::ExpandToEncompass(const BDSExtent& other)
236{
237 BDSExtent result = BDSExtent(*this);
238 result.extXNeg = std::min(extXNeg, other.extXNeg);
239 result.extYNeg = std::min(extYNeg, other.extYNeg);
240 result.extZNeg = std::min(extZNeg, other.extZNeg);
241 result.extXPos = std::max(extXPos, other.extXPos);
242 result.extYPos = std::max(extYPos, other.extYPos);
243 result.extZPos = std::max(extZPos, other.extZPos);
244}
