241void Parameters::inherit_properties(const Element& e)
242{
243 // copy parameters into temporary buffer params from element e
244 // parameters already set in params have priority and are not overridden
245 // this is used for the inheritance / newinstance mechanism
246
247 for (auto& i : setMap)
248 {
249 if(i.second == false)
250 {
251 std::string property = i.first;
252 // method can in theory throw runtime_error (shouldn't happen), catch and exit gracefully
253 try
254 {Published<Element>::set(this,(Element*)&e,property);}
255 catch(const std::runtime_error&)
256 {
257 std::cerr << "Error: element> unknown property \"" << property
258 << "\" from element " << e.name << std::endl;
259 exit(1);
260 }
261 i.second = true;
262 }
263 }
