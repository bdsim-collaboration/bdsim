268 if (nextLinkedVertex)

301 const std::vector<BDSHitsCollectionSamplerCylinder*>& samplerHitsCylinder,
302 const std::vector<BDSHitsCollectionSamplerSphere*>& samplerHitsSphere,

354 FillPrimaryLoss(primaryLosses);
355 if (trajectories)
356 {FillTrajectories(trajectories);}
357 if (collimatorHits)
358 {FillCollimatorHits(collimatorHits, primaryLosses);}

361 FillScorerHits(scorerHits); // map always exists
362
363 // we do this after energy loss and collimator hits as the energy loss
364 // is integrated for putting in event info and the number of collimators
365 // interacted with counted
366 if (info)
367 {FillEventInfo(info);}
368
369 WriteFileEventLevel();
370 ClearStructuresEventLevel();
371}

406G4String BDSOutput::GetNextFileName()
407{
408 outputFileNumber++;
409 const BDSGlobalConstants* globalConstants = BDSGlobalConstants::Instance();
410
411 // Base root file name
412 G4String newFileName = baseFileName;
413
414 // if more than one file add number (starting at 0)
415 // of numberEventPerFile is specified and the number already generated exceeds that
416 if (numberEventPerFile > 0 && globalConstants->NGenerate() > numberEventPerFile)
417 {newFileName += "_" + std::to_string(outputFileNumber);} // note underscore
418
419 // policy: overwrite if output filename specifically set, otherwise increase number
420 // always check in interactive mode
421 if (!globalConstants->OutputFileNameSet() || !globalConstants->Batch())
422 {// check if file exists
423 G4String original = newFileName; // could have nper file number suffix too
424 G4int nTimeAppended = 1;
425 while (BDS::FileExists(newFileName + fileExtension)) // always test with extension
426 {// if exists increment suffix integer
427 newFileName = original + "-" + std::to_string(nTimeAppended);
428 nTimeAppended +=1;
429 }
430 }
431
432 // add extension now we've got the base part fixed
433 newFileName += fileExtension;
434
435 G4cout << __METHOD_NAME__ << "Setting up new file: " << newFileName << G4endl;
436
437 return newFileName;
438}

440void BDSOutput::CalculateHistogramParameters()
441{
442 // rounding up so last bin definitely covers smax
443 // (max - min) / bin width -> min = 0 here.
444 const G4double binWidth = BDSGlobalConstants::Instance()->ELossHistoBinWidth();
445 const BDSBeamline* flatBeamline = BDSAcceleratorModel::Instance()->BeamlineMain();
446 if (flatBeamline)
447 {// don't access a nullptr
448 if (!flatBeamline->empty())
449 {
450 G4double sMax = flatBeamline->GetLastItem()->GetSPositionEnd();
451 nbins = (int) std::ceil((sMax - sMinHistograms)/ binWidth); // round up to integer # of bins
452 }
453 }
454 else
455 {nbins = 1;} // can happen for generate primaries only
456
457 if (nbins == 0)
458 {nbins = 1;}
459
460 sMaxHistograms = sMinHistograms + nbins * binWidth;
461}

463void BDSOutput::CreateHistograms()
464{
465 // construct output histograms
466 // calculate histogram dimensions
467 CalculateHistogramParameters();
468 const G4double smin = sMinHistograms / CLHEP::m;
469 const G4double smax = sMaxHistograms / CLHEP::m;
470#ifdef BDSDEBUG
471 G4cout << __METHOD_NAME__ << "histogram parameters calculated to be: " << G4endl;
472 G4cout << "s minimum: " << smin << " m" << G4endl;
473 G4cout << "s maximum: " << smax << " m" << G4endl;
474 G4cout << "# of bins: " << nbins << G4endl;
475#endif
476 // create the histograms
477 if (storePrimaryHistograms)
478 {
479 histIndices1D["Phits"] = Create1DHistogram("PhitsHisto","Primary Hits",nbins,smin,smax);
480 histIndices1D["Ploss"] = Create1DHistogram("PlossHisto","Primary Loss",nbins,smin,smax);
481 }
482 if (storeELossHistograms)
483 {histIndices1D["Eloss"] = Create1DHistogram("ElossHisto","Energy Loss",nbins,smin,smax);}
484 // prepare bin edges for a by-element histogram
485 std::vector<G4double> binedges;
486 const BDSBeamline* flatBeamline = BDSAcceleratorModel::Instance()->BeamlineMain();
487 if (flatBeamline) // can be nullptr in case of generate primaries only
488 {binedges = flatBeamline->GetEdgeSPositions();}
489 else
490 {binedges = {0,1};}
491 // create per element ("pe") bin width histograms
492 if (storePrimaryHistograms)
493 {
494 histIndices1D["PhitsPE"] = Create1DHistogram("PhitsPEHisto",
495 "Primary Hits per Element",
496 binedges);
497 histIndices1D["PlossPE"] = Create1DHistogram("PlossPEHisto",
498 "Primary Loss per Element",
499 binedges);
500 }
501 if (storeELossHistograms)
502 {
503 histIndices1D["ElossPE"] = Create1DHistogram("ElossPEHisto",
504 "Energy Loss per Element" ,
505 binedges);
506 }
507 if (storeELossVacuumHistograms)
508 {
509 histIndices1D["ElossVacuum"] = Create1DHistogram("ElossVacuumHisto",
510 "Energy Loss in Vacuum",
511 nbins,smin,smax);
512 histIndices1D["ElossVacuumPE"] = Create1DHistogram("ElossVaccumPEHisto",
513 "Energy Loss in Vacuum per Element" ,
514 binedges);
515 }
516
517 if (storeApertureImpactsHistograms)
518 {
519 histIndices1D["PFirstAI"] = Create1DHistogram("PFirstAIHisto",
520 "Primary aperture impacts",
521 nbins, smin, smax);
522 }
523
524 // only create tunnel histograms if we build the tunnel
525 const BDSBeamline* tunnelBeamline = BDSAcceleratorModel::Instance()->TunnelBeamline();
526 if (!tunnelBeamline)
527 {
528 storeELossTunnel = false;
529 storeELossTunnelHistograms = false;
530 }
531 if (storeELossTunnelHistograms && tunnelBeamline)
532 {
533 binedges = tunnelBeamline->GetEdgeSPositions();
534 histIndices1D["ElossTunnel"] = Create1DHistogram("ElossTunnelHisto",
535 "Energy Loss in Tunnel",
536 nbins, smin,smax);
537 histIndices1D["ElossTunnelPE"] = Create1DHistogram("ElossTunnelPEHisto",
538 "Energy Loss in Tunnel per Element",
539 binedges);
540 }
541
542 if (storeCollimatorInfo && nCollimators > 0)
543 {
544 std::vector<G4String> collHistNames = {"CollPhitsPE",
545 "CollPlossPE",
546 "CollElossPE",
547 "CollPInteractedPE"};
548 std::vector<G4String> collHistDesciptions = {"Primary Hits per Coll",
549 "Primary Loss per Coll",
550 "Energy Loss per Collimator",
551 "Primary Interacted per Collimator"};
552 for (G4int i = 0; i < (G4int)collHistNames.size(); i++)
553 {
554 histIndices1D[collHistNames[i]] = Create1DHistogram(collHistNames[i],
555 collHistDesciptions[i],
556 nCollimators, 0, nCollimators);
557 }
558 }
559
560 // one unique 'scorer' - single 3d histogram 3d
561 if (useScoringMap && storeELossHistograms)
562 {
563 const BDSGlobalConstants* g = BDSGlobalConstants::Instance();
564 if (!BDS::IsFinite(g->XMax() - g->XMin()))
565 {throw BDSException(__METHOD_NAME__, "0 width in general 3D scoring histogram in x dimension - check options, xmin and xmax");}
566 if (!BDS::IsFinite(g->YMax() - g->YMin()))
567 {throw BDSException(__METHOD_NAME__, "0 width in general 3D scoring histogram in y dimension - check options, ymin and ymax");}
568 if (!BDS::IsFinite(g->ZMax() - g->ZMin()))
569 {throw BDSException(__METHOD_NAME__, "0 width in general 3D scoring histogram in z dimension - check options, zmin and zmax");}
570 if (g->NBinsX() <= 0)
571 {throw BDSException(__METHOD_NAME__, "invalid number of bins in x dimension of 3D scoring histogram - check option, nbinsx");}
572 if (g->NBinsY() <= 0)
573 {throw BDSException(__METHOD_NAME__, "invalid number of bins in y dimension of 3D scoring histogram - check option, nbinsx");}
574 if (g->NBinsZ() <= 0)
575 {throw BDSException(__METHOD_NAME__, "invalid number of bins in z dimension of 3D scoring histogram - check option, nbinsx");}
576
577 G4int scInd = Create3DHistogram("ScoringMap", "Energy Deposition",
578 g->NBinsX(), g->XMin()/CLHEP::m, g->XMax()/CLHEP::m,
579 g->NBinsY(), g->YMin()/CLHEP::m, g->YMax()/CLHEP::m,
580 g->NBinsZ(), g->ZMin()/CLHEP::m, g->ZMax()/CLHEP::m);
581 histIndices3D["ScoringMap"] = scInd;
582 }
583
584 // scoring maps
585 const std::map<G4String, BDSScorerHistogramDef> scorerHistogramDefs = BDSAcceleratorModel::Instance()->ScorerHistogramDefinitionsMap();
586 if (!scorerHistogramDefs.empty())
587 {
588 for (const auto& nameDef : scorerHistogramDefs)
589 {
590 const auto def = nameDef.second;
591 // use safe output name without any slashes in the name
592 G4int histID = -1;
593
594 if (def.nBinsE <=1)
595 {
596 if (def.geometryType == "box")
597 {
598 histID = Create3DHistogram(def.outputName, def.outputName,
599 def.nBinsX, def.xLow/CLHEP::m, def.xHigh/CLHEP::m,
600 def.nBinsY, def.yLow/CLHEP::m, def.yHigh/CLHEP::m,
601 def.nBinsZ, def.zLow/CLHEP::m, def.zHigh/CLHEP::m);
602 }
603 else if (def.geometryType == "cylindrical")
604 {
605 histID = Create3DHistogram(def.outputName, def.outputName,
606 def.nBinsZ, def.zLow/CLHEP::m, def.zHigh/CLHEP::m,
607 def.nBinsPhi, 0, CLHEP::twopi,
608 def.nBinsR, def.rLow/CLHEP::m, def.rHigh/CLHEP::m);
609 }
610
611 histIndices3D[def.uniqueName] = histID;
612 histIndexToUnits3D[histID] = def.primitiveScorerUnitValue;
613 // avoid using [] operator for map as we have no default constructor for BDSHistBinMapper3D
614 }
615 else
616 {
617 if (def.geometryType == "box")
618 {
619 histID = Create4DHistogram(def.outputName+"-"+def.eScale,def.outputName,def.eScale,def.eBinsEdges,
620 def.nBinsX, def.xLow/CLHEP::m, def.xHigh/CLHEP::m,
621 def.nBinsY, def.yLow/CLHEP::m, def.yHigh/CLHEP::m,
622 def.nBinsZ, def.zLow/CLHEP::m, def.zHigh/CLHEP::m,
623 def.nBinsE, def.eLow/CLHEP::GeV, def.eHigh/CLHEP::GeV);

632 }

635 histIndexToUnits4D[histID] = def.primitiveScorerUnitValue;
636 }
637 scorerCoordinateMaps.insert(std::make_pair(def.uniqueName, def.coordinateMapper));
638 }
639 }
640
641 G4int nBLMs = BDSBLMRegistry::Instance()->NBLMs();
642 if (nBLMs > 0)
643 {
644 // the same primitive scorers for BLMs may be used in multiple SDs (each representing
645 // a unique combination of primitive scorers. However, we want 1 histogram per primitive
646 // scorer for all BLMs. We want to fill the same scoring quantity into the one histogram
647 // even from different collections ("SD/PS"). Determine 'set' of histogram names, which is
648 // set of primitive scorers use for BLMs.
649 // note there may be scorers from general 3d meshes and not just blms
650 std::vector<G4String> psnamesc = BDSSDManager::Instance()->PrimitiveScorerNamesComplete();
651 std::vector<G4String> psnames = BDSSDManager::Instance()->PrimitiveScorerNames();
652 std::map<G4String, G4double> scorerUnits = BDSSDManager::Instance()->PrimitiveScorerUnits();
653 std::set<G4String> blmHistoNames;
654 std::map<G4String, G4String> psFullNameToPS;
655 for (const auto& scorerNameComplete : psnamesc)
656 {
657 if (BDS::StrContains(scorerNameComplete, "blm_"))
658 {
659 for (const auto& scorerName : psnames)
660 {
661 if (BDS::StrContains(scorerNameComplete, "/"+scorerName)) // only match end of full name with '/'
662 {
663 blmHistoNames.insert(scorerName);
664 psFullNameToPS[scorerNameComplete] = scorerName;
665 }
666 }
667 }
668 }

670 // make BLM histograms and map the full collection name to that histogram ID for easy filling
671 // at the end of event. Note, multiple collections may feed into the same histogram.
672 for (const auto &hn : blmHistoNames)
673 {
674 G4String blmHistName = "BLM_" + hn;
675 G4int hind = Create1DHistogram(blmHistName, blmHistName, nBLMs, 0, nBLMs);
676 histIndices1D[blmHistName] = hind;
677 histIndexToUnits1D[hind] = scorerUnits[hn];
678 for (const auto& kv : psFullNameToPS)
679 {
680 if (hn == kv.second)
681 {blmCollectionNameToHistogramID[kv.first] = hind;}
682 }
683 }

1039void BDSOutput::FillCollimatorHits(const BDSHitsCollectionCollimator* hits,
1040 const std::vector<const BDSTrajectoryPointHit*>& primaryLossPoints)
1041{
1042 G4int nHits = (G4int)hits->entries();
1043 for (G4int i = 0; i < nHits; i++)
1044 {
1045 BDSHitCollimator* hit = (*hits)[i];
1046 G4int collimatorIndex = hit->collimatorIndex;
1047 collimators[collimatorIndex]->Fill(hit,
1048 collimatorInfo[collimatorIndex],
1049 collimatorDifferences[collimatorIndex],
1050 storeCollimatorHits); // this includes the || storeCollimatorHitsLinks || storeCollimatorHitsIons);
1051 }
1052
1053 // identify whether the primary loss point was in a collimator
1054 // only do if there's a beam line, ie finished in a beam line, and there are collimators
1055 if (!primaryLossPoints.empty())
1056 {
1057 for (auto primaryLossPoint : primaryLossPoints)
1058 {
1059 if (primaryLossPoint->point->GetBeamLine() && nCollimators > 0)
1060 {
1061 G4int lossPointBLInd = primaryLossPoint->point->GetBeamLineIndex(); // always the mass world index
1062 auto result = std::find(collimatorIndices.begin(), collimatorIndices.end(), lossPointBLInd);
1063 if (result != collimatorIndices.end())
1064 {
1065 G4int collIndex = (int) (result - collimatorIndices.begin());
1066 collimators[collIndex]->SetPrimaryStopped(true);
1067 collimators[collIndex]->primaryInteracted = true;
1068 // it must've interacted if it stopped - could be that we kill
1069 // secondaries and there's no energy deposition therefore not identified
1070 // as primaryInteracted=true in BDSOutputROOTEventCollimator::Fill()
1071 }
1072 }
1073 }
1074 }
1075
1076 // if required loop over collimators and get them to calculate and fill extra information
1077 if (storeCollimatorHitsLinks || storeCollimatorHitsIons)
1078 {
1079 for (auto collimator : collimators)
1080 {collimator->FillExtras(storeCollimatorHitsIons, storeCollimatorHitsLinks);}
1081 }
1082
1083 // after all collimator hits have been filled, we summarise whether the primary
1084 // interacted in a histogram
1085 G4int histIndex = histIndices1D["CollPInteractedPE"];
1086 for (G4int i = 0; i < (G4int)collimators.size(); i++)
1087 {evtHistos->Fill1DHistogram(histIndex, i, (int)collimators[i]->primaryInteracted);}
1088
1089
1090 // loop over collimators and count the number that were interacted with in this event
1091 for (const auto collimator : collimators)
1092 {
1093 if (collimator->primaryInteracted)
1094 {nCollimatorsInteracted += 1;}
1095 }
1096}
