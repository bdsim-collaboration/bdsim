109void PhysicsBiasing::set_value(const std::string& property, std::string value, bool bExit)
110{
111#ifdef BDSDEBUG
112 std::cout << "parser> Setting value " << std::setw(25) << std::left << property << value << std::endl;
113#endif
114
115 if (property=="name")
116 {name = value;}
117 else if (property=="particle")
118 {particle = value;}
119 else if ((property=="proc") || (property=="process"))
120 {
121 process = value;
122 std::stringstream ss(process);
123 std::string tok;
124 while(ss >> tok)
125 {processList.push_back(tok);}
126 return;
127 }
128 else
129 {
130 std::cerr << "Error: parser> unknown physicsbiasing option \"" << property
131 << "\" with value " << value << std::endl;
132 if(bExit)
133 exit(1);
134 else
135 std::rethrow_exception(std::current_exception());
136 }
137}
