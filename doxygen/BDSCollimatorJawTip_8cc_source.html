72void BDSCollimatorJawTip::UpdateCalculations()
73{
74 BDSCollimatorJaw::Calculations();
75
76 // shrink the bulk jaws from the aperture side to make space for the tips, so the
77 // centre of each jaw moves outwards by half the space taken
78 G4double tipSpace = tipThickness + 2*lengthSafety;
79 leftJawWidth -= tipSpace;
80 leftJawPos += G4ThreeVector(0.5*tipSpace, 0, 0);
81 rightJawWidth -= tipSpace;
82 rightJawPos -= G4ThreeVector(0.5*tipSpace, 0, 0);
83
84 leftJawTipPos = G4ThreeVector(leftJawHalfGap + 0.5*tipThickness, 0, 0);
85 rightJawTipPos = G4ThreeVector(-(rightJawHalfGap + 0.5*tipThickness), 0, 0);
86}

88void BDSCollimatorJawTip::CheckParametersForTips()
89{
90 // the remaining checks only apply to the jaw and vacuum geometry
91 if (!buildAperture)
92 {
93 if (tipThickness > (0.5*horizontalWidth-4*lengthSafety))
94 {throw BDSException(__METHOD_NAME__, "2x tipThickness is greater than the horizontal width for zero opening jcoltip \"" + name + "\"");}
95 return;
96 }
97
98 // tip solids have a half width of tipThickness/2 - lengthSafety
99 if (tipThickness * 0.5 - lengthSafety < 1e-3) // 1um minimum, could also be negative
100 {throw BDSException(__METHOD_NAME__, "insufficient tipThickness for jcoltip \"" + name + "\"");}
101
102 // bulk jaw solids have a half width of (jawWidth - tipThickness)/2 - lengthSafety
103 if (buildLeftJaw && ((0.5 * horizontalWidth - leftJawHalfGap - tipThickness) * 0.5 - lengthSafety < 1e-3))
104 {throw BDSException(__METHOD_NAME__, "left jaw of jcoltip \"" + name + "\" too thin given horizontalWidth, aperture and tipThickness");}
105 if (buildRightJaw && ((0.5 * horizontalWidth - rightJawHalfGap - tipThickness) * 0.5 - lengthSafety < 1e-3))
106 {throw BDSException(__METHOD_NAME__, "right jaw of jcoltip \"" + name + "\" too thin given horizontalWidth, aperture and tipThickness");}
107}

116void BDSCollimatorJawTip::BuildTips()
117{
118 G4VisAttributes* tipVisAttr = new G4VisAttributes(*tipColour);
119 RegisterVisAttributes(tipVisAttr);
120
121 G4UserLimits* tipCollUserLimits = CollimatorUserLimits();
122
123 // If not aperture, then the base class just fills the container with one material. Now,
124 // place a tip material block inside that block.
125 if (!buildAperture && tipThickness > 1e-3)
126 {
127 G4VSolid* tipSolid = new G4Box(name + "_tip_solid",
128 tipThickness,
129 yHalfHeight - 2*lengthSafety,
130 chordLength * 0.5 - 2*lengthSafety);
131 // the containerLV is of length chordLength * 0.5 - lengthSafety
132 RegisterSolid(tipSolid);
133
134 G4LogicalVolume* collimatorTipLV = new G4LogicalVolume(tipSolid, collimatorTipMaterial, name + "_tip_lv");
135 collimatorTipLV->SetVisAttributes(tipVisAttr);
136 collimatorTipLV->SetUserLimits(tipCollUserLimits);
137
138 // register with base class (BDSGeometryComponent)
139 RegisterLogicalVolume(collimatorTipLV);
140 BDSAcceleratorModel::Instance()->VolumeSet("collimators")->insert(collimatorTipLV);
141 if (sensitiveOuter)
142 {RegisterSensitiveVolume(collimatorTipLV, BDSSDType::collimatorcomplete);}
143
144 // place the jaw
145 G4PVPlacement* collimatorTipPV = new G4PVPlacement(nullptr, // rotation
146 (G4ThreeVector) 0, // position
147 collimatorTipLV, // its logical volume
148 name + "_tip_pv", // its name
149 collimatorLV, // its mother volume
150 false, // no boolean operation
151 0, // copy number
152 checkOverlaps);
153 RegisterPhysicalVolume(collimatorTipPV);
154 return;
155 }
156
157 G4VSolid* leftJawTipSolid = nullptr;
158 if (buildLeftJaw && buildAperture)
159 {
160 if (jawTiltLeft != 0)
161 {
162 G4double leftHalfLength = chordLength * 0.5 * std::cos(jawTiltLeft);
163 leftJawTipSolid = new G4Para(name + "_leftjawtip_solid",
164 tipThickness * 0.5 - lengthSafety,
165 yHalfHeight - lengthSafety,
166 leftHalfLength - lengthSafety,
167 0, jawTiltLeft, 0);
168 }
169 else
170 {
171 leftJawTipSolid = new G4Box(name + "_leftjawtip_solid",
172 tipThickness * 0.5 - lengthSafety,
173 yHalfHeight - lengthSafety,
174 chordLength * 0.5 - lengthSafety);
175 }
176 RegisterSolid(leftJawTipSolid);
177
178 G4LogicalVolume* leftJawTipLV = new G4LogicalVolume(leftJawTipSolid, collimatorTipMaterial, name + "_leftjawtip_lv");
179 leftJawTipLV->SetVisAttributes(tipVisAttr);
180 leftJawTipLV->SetUserLimits(tipCollUserLimits);
181 RegisterLogicalVolume(leftJawTipLV);
182 BDSAcceleratorModel::Instance()->VolumeSet("collimators")->insert(leftJawTipLV);
183 if (sensitiveOuter)
184 {RegisterSensitiveVolume(leftJawTipLV, BDSSDType::collimatorcomplete);}
185
186 // place the tip
187 G4PVPlacement* leftJawTipPV = new G4PVPlacement(nullptr,
188 leftJawTipPos,
189 leftJawTipLV,
190 name + "_leftjawtip_pv",
191 containerLogicalVolume,
192 false, // no boolean operation
193 0, // copy number
194 checkOverlaps);
195 RegisterPhysicalVolume(leftJawTipPV);
196 }
197
198 G4VSolid* rightJawTipSolid = nullptr;
199 if (buildRightJaw && buildAperture)
200 {
201 if (jawTiltRight != 0)
202 {
203 G4double rightHalfLength = chordLength * 0.5 * std::cos(jawTiltRight);
204 rightJawTipSolid = new G4Para(name + "_rightjawtip_solid",
205 tipThickness * 0.5 - lengthSafety,
206 yHalfHeight - lengthSafety,
207 rightHalfLength - lengthSafety,
208 0, jawTiltRight, 0);
209 }
210 else
211 {
212 rightJawTipSolid = new G4Box(name + "_rightjawtip_solid",
213 tipThickness * 0.5 - lengthSafety,
214 yHalfHeight - lengthSafety,
215 chordLength * 0.5 - lengthSafety);
216 }
217 RegisterSolid(rightJawTipSolid);
218
219 G4LogicalVolume* rightJawTipLV = new G4LogicalVolume(rightJawTipSolid, collimatorTipMaterial, name + "_rightjawtip_lv");
220 rightJawTipLV->SetVisAttributes(tipVisAttr);
221 rightJawTipLV->SetUserLimits(tipCollUserLimits);
222 RegisterLogicalVolume(rightJawTipLV);
223 BDSAcceleratorModel::Instance()->VolumeSet("collimators")->insert(rightJawTipLV);
224 if (sensitiveOuter)
225 {RegisterSensitiveVolume(rightJawTipLV, BDSSDType::collimatorcomplete);}
226
227 // place the tip
228 G4PVPlacement* rightJawTipPV = new G4PVPlacement(nullptr,
229 rightJawTipPos,
230 rightJawTipLV,
231 name + "_rightjawtip_pv",
232 containerLogicalVolume,
233 false, // no boolean operation
234 0, // copy number
235 checkOverlaps);
236 RegisterPhysicalVolume(rightJawTipPV);
237 }
