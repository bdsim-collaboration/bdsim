41G4ThreeVector BDSFieldMagInterpolated2D::GetField(const G4ThreeVector& position,
42 const G4double t) const
43{
44 G4double fCoordinate = 0;
45 if (firstTime)
46 {fCoordinate = t;}
47 else
48 {fCoordinate = position[firstDimensionIndex];}
49 G4double sCoordinate = 0;
50 if (secondTime)
51 {sCoordinate = t;}
52 else
53 {sCoordinate = position[secondDimensionIndex];}
54 return interpolator->GetInterpolatedValue(fCoordinate, sCoordinate) * Scaling();
55}
