86BDSGeometryExternal* BDSGeometryFactory::BuildGeometry(G4String componentName,
87 const G4String& formatAndFileName,
88 std::map<G4String, G4Colour*>* colourMapping,
89 G4bool autoColour,
90 G4double suggestedLength,
91 G4double suggestedHorizontalWidth,
92 std::vector<G4String>* namedVacuumVolumes,
93 G4bool makeSensitive,
94 BDSSDType sensitivityType,
95 BDSSDType vacuumSensitivityType,
96 G4bool stripOuterVolumeAndMakeAssembly,
97 G4UserLimits* userLimitsToAttachToAllLVs,
98 G4bool dontReloadGeometry)
99{
100 std::pair<G4String, G4String> ff = BDS::SplitOnColon(formatAndFileName);
101 G4String fileName = BDS::GetFullPath(ff.second);
102
103 G4String searchName = fileName;
104 // If we strip the outer volume we're technically modifying the geometry
105 // and should prepare a unique load of it and cache that otherwise, we
106 // will get either the original or the stripped version only if we place
107 // the load the same geometry twice with/without stripping
108 if (stripOuterVolumeAndMakeAssembly)
109 {searchName += "_stripped";}
110
111 // we have this option to purposively not reload geometry uniquely for multiple
112 // instances by different component names - we need this sometimes, but we must
113 // use it cautiously knowing all the instances will be identical
114 // therefore we use a fixed nonsense name that someone is prevented from calling
115 // a component (because it's an option)
116 G4String searchComponentName = componentName;
117 if (dontReloadGeometry)
118 {searchComponentName = "dontReloadGeometry";}
119 // we must still build it with component name in case it's preprocessed. If we used
120 // 'dontReloadGeometry' then variables / stuff in that file could be degeneratively
121 // named in other unique geometry files that are loaded.
122
123 auto nameAndField = std::make_pair((std::string)searchName, searchComponentName);
124 const auto search = registry.find(nameAndField);
125 if (search != registry.end())
126 {return search->second;}// it was found already in registry
127 // else wasn't found so continue
128
129 // Check the file exists.
130 if (!BDS::FileExists(fileName))
131 {throw BDSException(__METHOD_NAME__, "No such file \"" + fileName + "\"");}
132
133 BDSGeometryType format = BDS::DetermineGeometryType(ff.first);
134 BDSGeometryFactoryBase* factory = GetAppropriateFactory(format);
135 if (!factory)
136 {return nullptr;}
137
138 BDSGeometryExternal* result = factory->Build(componentName,
139 fileName,
140 colourMapping,
141 autoColour,
142 suggestedLength,
143 suggestedHorizontalWidth,
144 namedVacuumVolumes,
145 makeSensitive,
146 sensitivityType,
147 vacuumSensitivityType,
148 userLimitsToAttachToAllLVs);
149
150 if (result)
151 {
152 if (stripOuterVolumeAndMakeAssembly)
153 {result->StripOuterAndMakeAssemblyVolume();}
154
155 // cache using optionally modified name
156 auto key = std::make_pair((std::string)searchName, searchComponentName);
157 registry[key] = result;
158 storage.insert(result);
159 }
160
161 return result;
162}
