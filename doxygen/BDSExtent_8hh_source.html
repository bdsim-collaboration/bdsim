174inline G4bool BDSExtent::operator<(const BDSExtent& r) const
175{
176 G4bool xOK = (extXNeg > r.extXNeg) && (extXPos < r.extXPos);
177 G4bool yOK = (extYNeg > r.extYNeg) && (extYPos < r.extYPos);
178 G4bool zOK = (extZNeg > r.extZNeg) && (extZPos < r.extZPos);
179 return xOK && yOK && zOK;
180}
181
182inline G4bool BDSExtent::operator==(const BDSExtent& r) const
183{
184 G4bool xOK = (extXNeg == r.extXNeg) && (extXPos == r.extXPos);
185 G4bool yOK = (extYNeg == r.extYNeg) && (extYPos == r.extYPos);
186 G4bool zOK = (extZNeg == r.extZNeg) && (extZPos == r.extZPos);
187 return xOK && yOK && zOK;
188}

190inline G4bool BDSExtent::TransverselyLessThan(const BDSExtent& r) const
191{
192 G4bool xOK = (extXNeg > r.extXNeg) && (extXPos < r.extXPos);
193 G4bool yOK = (extYNeg > r.extYNeg) && (extYPos < r.extYPos);
194 return xOK && yOK;
195}
