71 defines["kapton"] = c->GetColour("kapton");
72 defines["lead"] = c->GetColour("lead");
73 defines["pb"] = defines["lead"];
74 defines["lyso"] = c->GetColour("lyso: 230 210 235 0.3");
75 defines["lso"] = defines["lyso"];
76 defines["yso"] = defines["lyso"];
77 defines["lysoce"] = defines["lyso"];
78 defines["marble"] = c->GetColour("marble:228 228 228 1.0");
79 defines["stainlesssteel"] = c->GetColour("beampipe");
80 defines["stainless-steel"] = defines["stainlesssteel"];
81 defines["sulphur"] = c->GetColour("yellow");
82 defines["s"] = defines["sulphur"];
83 defines["vacuum"] = defines["air"];
84 defines["cucrzr"] = defines["copper"];
85
86 // for older versions of Geant4 < V11 we have to use G4DataVector which
87 // can't use list initialisation. In V11 onwards, G4PhysicsFreeVector
88 // changed to take std::vector<G4double> as arguments but G4DataVector
89 // inherits this so it's ok. Order must be ascending.
90 G4DataVector densities(5);
91 for (auto v : {1e-4, 0.01, 0.1, 1.0, 1e2})
92 {densities.emplace_back(v * CLHEP::g / CLHEP::cm3);}
93 G4DataVector values(5);
94 for (auto v : {210, 180, 150, 120, 100})
95 {values.emplace_back(v);}
96 generalDensity = new G4PhysicsFreeVector(densities, values);
97}

99G4Colour* BDSColourFromMaterial::GetColour(const G4Material* material,
100 const G4String& prefixToStripFromName)
101{
102 G4String materialName = material->GetName();
103 materialName = BDS::LowerCase(materialName);
104
105 // strip off g4 so we don't have to define duplicates of everything
106 std::string toErase = "g4_";
107 size_t pos = materialName.find(toErase);
108 if (pos != std::string::npos)
109 {materialName.erase(pos, toErase.length());}
110
111 // note, not all material are prefixed by preprocessor and might not contain that prefix string
112 G4String prefixToStripFromNameLower = BDS::LowerCase(prefixToStripFromName);
113 if (!prefixToStripFromName.empty() && BDS::StrContains(materialName, prefixToStripFromNameLower))
114 {materialName.erase(0, prefixToStripFromName.size());}
115
116 auto search = defines.find(materialName);
117 if (search != defines.end())
118 {return search->second;}
119 else
120 {
121 G4double alpha = 1;
122 G4State state = material->GetState();
123 switch (state)
124 {
125 case G4State::kStateGas:
126 {alpha = 0.05; break;}
127 case G4State::kStateSolid:
128 {alpha = 1.0; break;}
129 case G4State::kStateLiquid:
130 {alpha = 0.5; break;}
131 default:
132 {alpha = 1.0; break;}
133 }
134 G4double density = material->GetDensity();
135 G4double value = generalDensity->Value(density);
136 G4String vs = G4String(std::to_string(value));
137 G4Colour* result = BDSColours::Instance()->GetColour(materialName + ":" +
138 vs + " " +
139 vs + " " +
140 vs + " " +
141 std::to_string(alpha));
142 defines[materialName] = result; // cache it as we'd always produce the same result
143 return result;
144 }
145}

147G4Colour* BDSColourFromMaterial::GetColourWithDefault(const G4Material* material,
148 G4Colour* defaultIn) const
149{
150 G4String materialName = material->GetName();
151 materialName = BDS::LowerCase(materialName);
152
153 // strip off g4 so we don't have to define duplicates of everything
154 std::string toErase = "g4_";
155 size_t pos = materialName.find(toErase);
156 if (pos != std::string::npos)
157 {materialName.erase(pos, toErase.length());}
158

160 return search != defines.end() ? search->second : defaultIn;
