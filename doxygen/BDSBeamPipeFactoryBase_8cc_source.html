109void BDSBeamPipeFactoryBase::SetVisAttributes(G4Material* beamPipeMaterialIn,
110 G4Material* vacuumMaterialIn)
111{
112 auto cfm = BDSColourFromMaterial::Instance();
113 G4Colour* c = cfm->GetColourWithDefault(beamPipeMaterialIn, BDSColours::Instance()->GetColour("beampipe"));
114 G4VisAttributes* pipeVisAttr = new G4VisAttributes(*c);
115 pipeVisAttr->SetVisibility(true);
116 pipeVisAttr->SetForceLineSegmentsPerCircle(nSegmentsPerCircle);
117 allVisAttributes.insert(pipeVisAttr);
118 beamPipeLV->SetVisAttributes(pipeVisAttr);
119
120 if (vacuumMaterialIn->GetDensity() > (1e-3*CLHEP::gram/CLHEP::cm3))
121 {
122 G4Colour* cv = cfm->GetColour(vacuumMaterialIn);
123 G4VisAttributes* vacVisAttr = new G4VisAttributes(*cv);
124 vacVisAttr->SetVisibility(true);
125 vacVisAttr->SetForceLineSegmentsPerCircle(nSegmentsPerCircle);
126 allVisAttributes.insert(vacVisAttr);
127 vacuumLV->SetVisAttributes(vacVisAttr);
128 }
129 else
130 {vacuumLV->SetVisAttributes(BDSGlobalConstants::Instance()->ContainerVisAttr());}
131
132 containerLV->SetVisAttributes(BDSGlobalConstants::Instance()->ContainerVisAttr());
133}

135void BDSBeamPipeFactoryBase::SetUserLimits(G4double length)
136{
137 auto defaultUL = BDSGlobalConstants::Instance()->DefaultUserLimits();
138 //copy the default and update with the length of the object rather than the default 1m
139 G4UserLimits* ul = BDS::CreateUserLimits(defaultUL, length);
140 if (ul != defaultUL) // if it's not the default register it
141 {allUserLimits.insert(ul);}
142 vacuumLV->SetUserLimits(ul);
143 containerLV->SetUserLimits(ul);
144
145 G4UserLimits* beamPipeUL = ul;
146 if (BDSGlobalConstants::Instance()->BeamPipeIsInfiniteAbsorber())
147 {// new beam pipe user limits, copy from updated default user limits above
148 beamPipeUL = new G4UserLimits(*ul);
149 beamPipeUL->SetUserMinEkine(std::numeric_limits<double>::max());
150 allUserLimits.insert(beamPipeUL);
151 }
152 beamPipeLV->SetUserLimits(beamPipeUL);
153}

181BDSBeamPipe* BDSBeamPipeFactoryBase::BuildBeamPipeAndRegisterVolumes(BDSExtent extent,
182 G4double containerRadius,
183 G4bool containerIsCircular)
184{
185 // build the BDSBeamPipe instance and return it
186 BDSBeamPipe* aPipe = new BDSBeamPipe(containerSolid,containerLV,extent,
187 containerSubtractionSolid,
188 vacuumLV,containerIsCircular,containerRadius,
189 inputFaceNormal, outputFaceNormal);
190
191 // register objects
192 aPipe->RegisterSolid(allSolids);
193 aPipe->RegisterLogicalVolume(allLogicalVolumes); //using geometry component base class method
194 aPipe->RegisterPhysicalVolume(allPhysicalVolumes);
195 if (sensitiveVacuum)
196 {aPipe->RegisterSensitiveVolume(vacuumLV, BDSSDType::energydepvacuum);}
197 if (beamPipeLV)
198 {
199 if (sensitiveBeamPipe && storeApertureImpacts)// in the case of the circular vacuum, there isn't a beampipeLV
200 {aPipe->RegisterSensitiveVolume(beamPipeLV, BDSSDType::aperturecomplete);}
201 else if (storeApertureImpacts)
202 {aPipe->RegisterSensitiveVolume(beamPipeLV, BDSSDType::apertureimpacts);}
203 else
204 {aPipe->RegisterSensitiveVolume(beamPipeLV, BDSSDType::energydep);}
205 }
206 aPipe->RegisterUserLimits(allUserLimits);
207 aPipe->RegisterVisAttributes(allVisAttributes);
208
209 return aPipe;
210}
