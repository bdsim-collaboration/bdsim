130void SamplerPlacement::set_value(const std::string& property, Array* value, bool bExit)
131{
132#ifdef BDSDEBUG
133 std::cout << "parser> Setting value " << std::setw(25) << std::left << property << std::endl;
134#endif
135 if(property=="partID")
136 {
137 value->set_vector(partID);
138 }
139 else
140 {
141 std::cerr << "Error: parser> unknown material option \"" << property << "\", or doesn't expect vector type" << std::endl;
142 if(bExit)
143 {exit(1);}
144 else
145 {std::rethrow_exception(std::current_exception());} // to be caught by python
146 }
147}
