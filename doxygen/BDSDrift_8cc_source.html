37void BDSDrift::Build()
38{
39 BDSBeamPipeFactory* factory = BDSBeamPipeFactory::Instance();
40 BDSBeamPipe* pipe = factory->CreateBeamPipe(name,
41 chordLength,
42 beamPipeInfo);
43
44 RegisterDaughter(pipe);
45
46 // make the beam pipe container, this object's container
47 containerLogicalVolume = pipe->GetContainerLogicalVolume();
48 containerSolid = pipe->GetContainerSolid();
49
50 // register vacuum volume (for biasing)
51 SetAcceleratorVacuumLogicalVolume(pipe->GetVacuumLogicalVolume());
52
53 // update extents
54 InheritExtents(pipe);
55
56 // update faces
57 SetInputFaceNormal(pipe->InputFaceNormal());
58 SetOutputFaceNormal(pipe->OutputFaceNormal());
59}
