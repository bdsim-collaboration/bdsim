57BDSBeamline* BDS::BuildBLMs(const std::vector<GMAD::BLMPlacement>& blmPlacements,
58 const BDSBeamline* parentBeamLine)
59{
60 if (blmPlacements.empty())
61 {return nullptr;} // don't do anything - no placements
62
63 // loop over blm placements - if any have external geometry, force the loading of it now so any materials
64 // that could be used in the BDSScorerInfo filters will be defined first by the load. The geometry is cached
65 // so it won't be loaded twice.
66 for (const auto& bp : blmPlacements)
67 {
68 if (!bp.geometryFile.empty())
69 {BDSGeometryFactory::Instance()->BuildGeometry(G4String(bp.name), G4String(bp.geometryFile));}
70 }
71
72 // we need to loop over all the blm definitions to work out the unique combinations of
73 // scorers that need to be created. multiple scorers for a single blm ultimately have to be
74 // in one G4MultiFunctionalSD sensitive detector.
75 std::vector<GMAD::Scorer> scorers = BDSParser::Instance()->GetScorers();
76 // convert all the parser scorer definitions into recipes (including parameter checking)
77 std::map<G4String, BDSScorerInfo> scorerRecipes;
78 for (const auto& scorer : scorers)
79 {
80 BDSScorerInfo si = BDSScorerInfo(scorer);
81 scorerRecipes.insert(std::make_pair(si.name, si));
82 }
83
84 std::set<std::set<G4String> > scorerSetsToMake;
85 // cache a map of set of scorers and combined name by BLM name so we don't have to do it later again
86 std::map<std::string, std::pair<std::set<G4String>, G4String> >blmScoringInfo;
87 for (const auto& bp : blmPlacements)
88 {
89 std::vector<G4String> scorersForThisBLM = BDS::SplitOnWhiteSpace(G4String(bp.scoreQuantity));
90 std::set<G4String> requiredScorers;
91 for (const auto& name : scorersForThisBLM)
92 {
93 auto search = scorerRecipes.find(name);
94 if (search == scorerRecipes.end())
95 {throw BDSException(__METHOD_NAME__, "scorerQuantity \"" + name + "\" for blm \"" + bp.name + "\" not found.");}
96 else
97 {requiredScorers.insert(name);}
98 }
99 // no BDS::Warning here as that slows down print out - could be all the blms turned off
100 if (requiredScorers.empty())
101 {G4cout << "Warning - no scoreQuantity specified for blm \"" << bp.name << "\" - it will only be passive material" << G4endl;}
102 scorerSetsToMake.insert(requiredScorers);
103
104 // the set by definition orders its contents, so we iterate through it to form a uniquely
105 // ordered combined name for the set.
106 G4String combinedName = "";
107 for (const auto& n : requiredScorers)
108 {combinedName += n;}
109 blmScoringInfo[bp.name] = std::make_pair(requiredScorers, combinedName);
110 }
111#ifdef BDSDEBUG
112 G4cout << "Unqiue combinations of scorers: " << scorerSetsToMake.size() << G4endl;
113#endif
114 if (scorerSetsToMake.empty())
115 {G4cout << "Warning - all BLMs have no scoreQuantity specified so are only passive material." << G4endl;}
116
117 // construct SDs
118 BDSScorerFactory scorerFactory;
119 std::map<G4String, G4MultiFunctionalDetector*> sensitiveDetectors;
120 std::vector<G4String> uniquePrimitiveScorerNames;
121 std::vector<G4double> scorerUnits;
122 G4SDManager* SDMan = G4SDManager::GetSDMpointer();
123 for (const auto& ssAndCombinedName : blmScoringInfo)
124 {
125 G4String combinedName = ssAndCombinedName.second.second;
126#ifdef BDSDEBUG
127 G4cout << __METHOD_NAME__ << "Making unique SD " << combinedName << G4endl;
128#endif
129 // If the sensitive detector with a unique combination of scorers is constructed already, skip
130 if (sensitiveDetectors.count(combinedName) > 0)
131 {continue;}
132
133 G4MultiFunctionalDetector* sd = new G4MultiFunctionalDetector("blm_"+combinedName);
134 for (const auto& name : ssAndCombinedName.second.first)
135 {
136 const auto& recipe = scorerRecipes.at(name); // safe as in previous loop we ensure it exists
137 G4double unit = 1.0;
138 G4VPrimitiveScorer* ps = scorerFactory.CreateScorer(&recipe, nullptr, &unit);
139 sd->RegisterPrimitive(ps);
140 // We rely on the prefix "blm_" here to intercept scorer hits in BDSOutput so if
141 // this changes, that matching must be done there too. It's to distinguish them
142 // from 3D mesh hits and put them in the BLM output.
143 uniquePrimitiveScorerNames.emplace_back("blm_"+combinedName+"/"+name);
144 scorerUnits.push_back(unit);
145 }
146 sensitiveDetectors[combinedName] = sd;
147 SDMan->AddNewDetector(sd);
148 }
149
150 BDSSDManager::Instance()->RegisterPrimitiveScorerNames(uniquePrimitiveScorerNames, &scorerUnits);
151
152 BDSBeamline* blms = new BDSBeamline();
153 BDSBLMFactory factory;
154 for (const auto& bp : blmPlacements)
155 {
156 // form a set of score quantities again
157 const auto& v = blmScoringInfo.at(bp.name);
158 std::set<G4String> requiredScorers = v.first;
159 G4String combinedName = v.second;
160
161 auto sdSearch = sensitiveDetectors.find(combinedName);
162 if (sdSearch == sensitiveDetectors.end())
163 {throw BDSException(__METHOD_NAME__, "unknown set of scorers");}
164 G4MultiFunctionalDetector* sd = sdSearch->second;
165
166 BDSBLM* blm = factory.CreateBLM(bp, sd);
167 BDSExtent blmExtent = blm->GetExtent();
168 G4double length = blmExtent.DZ();
169 if (!BDS::IsFinite(length))
170 {throw BDSException(__METHOD_NAME__, "BLM: " + bp.name + " has 0 extent");}
171 BDSSimpleComponent* comp = new BDSSimpleComponent(blm->GetName(),
172 blm,
173 length);
174 comp->Initialise();
175
176 G4double S = -1000;
177 G4Transform3D transform = BDSDetectorConstruction::CreatePlacementTransform(bp, parentBeamLine,
178 &S, &blmExtent);
179 // do a little checking here as transform code in CreatePlacement can't
180 // know who's calling it to warn
181 if (BDS::IsFinite(bp.s) && transform == G4Transform3D::Identity)
182 {
183 G4String msg = "the S coordinate (" + std::to_string(bp.s) + " m) for blm \"" + bp.name;
184 msg += "\" was not found in the beam line - please check";
185 throw BDSException(__METHOD_NAME__, msg);
186 }
187
188 BDSBLMRegistry::Instance()->RegisterBLM(bp.name, blm, S);
189
192 G4ThreeVector halfLengthBeg = G4ThreeVector(0,0,-length*0.5);
193 G4ThreeVector halfLengthEnd = G4ThreeVector(0,0, length*0.5);
194 G4ThreeVector midPos = transform.getTranslation();
195 G4ThreeVector startPos = midPos + transform * (HepGeom::Point3D<G4double>)halfLengthBeg;
196 G4ThreeVector endPos = midPos + transform * (HepGeom::Point3D<G4double>)halfLengthEnd;
197 G4RotationMatrix* rm = new G4RotationMatrix(transform.getRotation());
198
199 BDSBeamlineElement* el = new BDSBeamlineElement(comp,
200 startPos,
201 midPos,
202 endPos,
203 rm,
204 new G4RotationMatrix(*rm),
205 new G4RotationMatrix(*rm),
206 startPos,
207 midPos,
208 endPos,
209 new G4RotationMatrix(*rm),
210 new G4RotationMatrix(*rm),
211 new G4RotationMatrix(*rm),
212 -1,-1,-1, 0);
213
214 blms->AddBeamlineElement(el);
215 }
216
217 return blms;
218}
