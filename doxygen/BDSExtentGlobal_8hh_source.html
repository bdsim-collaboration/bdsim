144inline G4bool BDSExtentGlobal::TransverselyLessThanGlobal(const BDSExtentGlobal& r) const
145{
146 G4bool xOK = (extXNegG > r.extXNegG) && (extXPosG < r.extXPosG);
147 G4bool yOK = (extYNegG > r.extYNegG) && (extYPosG < r.extYPosG);
148 return xOK && yOK;
