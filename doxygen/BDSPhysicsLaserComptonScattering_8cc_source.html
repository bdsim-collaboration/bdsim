46void BDSPhysicsLaserComptonScattering::ConstructProcess()
47{
48 if (Activated())
49 {return;}
50
51 BDSLaserComptonScattering* laserComptonScattering = new BDSLaserComptonScattering();
52 G4AutoDelete::Register(laserComptonScattering);
53#if G4VERSION_NUMBER > 1029
54 auto aParticleIterator = GetParticleIterator();
55#endif
56 aParticleIterator->reset();
57 G4String name = laserComptonScattering->GetProcessName();
58 while ((*aParticleIterator)())
59 {
60 G4ParticleDefinition* particle = aParticleIterator->value();
61 G4int particleID = particle->GetPDGEncoding();
62 if (abs(particleID)==11)
63 {
64 G4ProcessManager* pmanager = particle->GetProcessManager();
65 pmanager->AddDiscreteProcess(laserComptonScattering);
66 }
67 if (particleID==18)
68 {
69 G4ProcessManager* pmanager = particle->GetProcessManager();
70 pmanager->AddDiscreteProcess(laserComptonScattering);
71 }
72 }
73
74 SetActivated();
75}
