190 template<class T>
191 T Linear1DMag(const T p[2],
192 G4double x)
193 {
194 T v = BDS::Linear1D(p,x);
195 double vMag = Linear1DMagOnly(p, x);
196 double factor = vMag / mag(v);
197 if (std::isnan(factor))
198 {factor = 1.0;}
199 v *= factor;
200 return v;
201 }
