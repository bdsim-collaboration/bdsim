342 void Element::set_value(std::string property, T value, bool exceptionSafe)
343 {
344#ifdef BDSDEBUG
345 std::cout << "element> Setting value " << std::setw(25) << std::left << property << value << std::endl;
346#endif
347 // member method can throw runtime_error, catch and exit gracefully
348 try
349 {Published<Element>::set(this, property, value);}
350 catch(const std::runtime_error&)
351 {
352 std::stringstream ss;
353 ss << "Error: element> unknown property \"" << property
354 << "\" with value \"" << value << "\" in definition \"" << this->name << "\"";
355 if (!exceptionSafe)
356 {throw std::invalid_argument(ss.str());}
357 else
358 {std::cerr << ss.str() << std::endl;}
359 }
360 }
