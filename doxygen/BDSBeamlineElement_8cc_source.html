41BDSBeamlineElement::BDSBeamlineElement(BDSAcceleratorComponent* componentIn,
42 const G4ThreeVector& positionStartIn,
43 const G4ThreeVector& positionMiddleIn,
44 const G4ThreeVector& positionEndIn,
45 G4RotationMatrix* rotationStartIn,
46 G4RotationMatrix* rotationMiddleIn,
47 G4RotationMatrix* rotationEndIn,
48 const G4ThreeVector& referencePositionStartIn,
49 const G4ThreeVector& referencePositionMiddleIn,
50 const G4ThreeVector& referencePositionEndIn,
51 G4RotationMatrix* referenceRotationStartIn,
52 G4RotationMatrix* referenceRotationMiddleIn,
53 G4RotationMatrix* referenceRotationEndIn,
54 G4double sPositionStartIn,
55 G4double sPositionMiddleIn,
56 G4double sPositionEndIn,
57 G4double synchronousTMiddleIn,
58 G4double startMomentumIn,
59 G4double startKineticEnergyIn,
60 BDSTiltOffset* tiltOffsetIn,
61 BDSSamplerInfo* samplerInfoIn,
62 G4int indexIn):
63 component(componentIn),
64 positionStart(positionStartIn), positionMiddle(positionMiddleIn), positionEnd(positionEndIn),
65 rotationStart(rotationStartIn), rotationMiddle(rotationMiddleIn), rotationEnd(rotationEndIn),
66 referencePositionStart(referencePositionStartIn),
67 referencePositionMiddle(referencePositionMiddleIn),
68 referencePositionEnd(referencePositionEndIn),
69 referenceRotationStart(referenceRotationStartIn),
70 referenceRotationMiddle(referenceRotationMiddleIn),
71 referenceRotationEnd(referenceRotationEndIn),
72 sPositionStart(sPositionStartIn), sPositionMiddle(sPositionMiddleIn), sPositionEnd(sPositionEndIn),
73 synchronousTMiddle(synchronousTMiddleIn),
74 startMomentum(startMomentumIn),
75 startKineticEnergy(startKineticEnergyIn),
76 tiltOffset(tiltOffsetIn),
77 samplerInfo(samplerInfoIn),
78 samplerPlacementTransform(nullptr),
79 index(indexIn)
80{
81 componentIn->IncrementCopyNumber(); // increase copy number (starts at -1)
82 copyNumber = componentIn->GetCopyNumber();
84 placementName = componentIn->GetName() + "_" + std::to_string(copyNumber);
85
86 // create the placement transform from supplied rotation matrices and vector
87 placementTransform = new G4Transform3D(*rotationMiddle, positionMiddle);
88 placementTransformCL = new G4Transform3D(*referenceRotationMiddle, referencePositionMiddle);
89
90 // calculate sampler central position slightly away from end position of element.
91 auto samplerType = GetSamplerType();
92 if (samplerType == BDSSamplerType::plane)
93 {
94 G4ThreeVector dZLocal = G4ThreeVector(0,0,1);
95 // if we have a sampler, add on the thickness of the sampler to the padding
96 dZLocal *= (BDSBeamline::PaddingLength() + BDSSamplerPlane::ChordLength())*0.5;
97 dZLocal.transform(*referenceRotationEnd);
98 G4ThreeVector samplerPosition = referencePositionEnd + dZLocal;
99 samplerPlacementTransform = new G4Transform3D(*referenceRotationEnd, samplerPosition);
100 }
101 else if (samplerType == BDSSamplerType::cylinder)
102 {samplerPlacementTransform = new G4Transform3D(*referenceRotationMiddle, referencePositionMiddle);}
103}

119std::set<G4VPhysicalVolume*> BDSBeamlineElement::GetPVsFromAssembly(G4AssemblyVolume* av)
120{
121 // this is obtuse because of the lack of access in G4Assembly
122 std::vector<G4VPhysicalVolume*> allPlacementsFromThisAssembly;
123 auto it = av->GetVolumesIterator();
124 for (G4int i = 0; i < (G4int)av->TotalImprintedVolumes(); it++, i++)
125 {allPlacementsFromThisAssembly.push_back(*it);}
126 std::set<G4VPhysicalVolume*> result(allPlacementsFromThisAssembly.begin(), allPlacementsFromThisAssembly.end());
127 return result;
128}

130std::set<G4VPhysicalVolume*> BDSBeamlineElement::PlaceElement(const G4String& pvName,
131 G4VPhysicalVolume* motherPV,
132 G4bool useCLPlacementTransform,
133 G4int pvCopyNumber,
134 G4bool checkOverlaps) const
135{
136 G4Transform3D* pvTransform = GetPlacementTransform();
137 if (useCLPlacementTransform)
138 {pvTransform = GetPlacementTransformCL();}
139
140 std::set<G4VPhysicalVolume*> result;
141 if (component->ContainerIsAssembly())
142 {
143 G4AssemblyVolume* av = component->GetContainerAssemblyVolume();
144 auto existingPVSet = GetPVsFromAssembly(av);
145 av->MakeImprint(motherPV->GetLogicalVolume(),
146 *pvTransform,
147 pvCopyNumber,checkOverlaps);
148
149 // need to get pvs before and afterwards and o the difference
150 auto updatedPVSet = GetPVsFromAssembly(av);
151 std::set_difference(updatedPVSet.begin(), updatedPVSet.end(),
152 existingPVSet.begin(), existingPVSet.end(),
153 std::inserter(result, result.begin()));
154 }
155 else
156 {
157 G4LogicalVolume* lv = component->GetContainerLogicalVolume();
158 result.insert(new G4PVPlacement(*pvTransform, // placement transform
159 pvName, // placement name
160 lv, // volume to be placed
161 motherPV, // volume to place it in
162 false, // no boolean operation
163 pvCopyNumber, // copy number
164 checkOverlaps)); // overlap checking
165 }
166 return result;
167}

176G4ThreeVector BDSBeamlineElement::InputFaceNormal() const
177{

220{
221 BDSExtentGlobal thisGlobal = BDSExtentGlobal(component->GetExtent(),
222 *(GetPlacementTransform()));
223 BDSExtentGlobal otherGlobal = BDSExtentGlobal(otherElement->GetAcceleratorComponent()->GetExtent(),
