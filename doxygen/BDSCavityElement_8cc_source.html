61void BDSCavityElement::Build()
62{
63 BDSCavity* cavity = BDSCavityFactory::Instance()->CreateCavity(name,
64 chordLength,
65 cavityInfo,
66 vacuumMaterial);
67 RegisterDaughter(cavity);
68
69 // make the beam pipe container, this object's container
70 containerLogicalVolume = cavity->GetContainerLogicalVolume();
71 containerSolid = cavity->GetContainerSolid();
72
73 // register vacuum volume (for biasing)
74 SetAcceleratorVacuumLogicalVolume(cavity->GetVacuumLogicalVolume());
75
76 // update extents
77 InheritExtents(cavity);
78
79 // field
80 BuildField();
81
82 BuildUserLimits();
83 AttachUserLimits();
84}

90 true);
